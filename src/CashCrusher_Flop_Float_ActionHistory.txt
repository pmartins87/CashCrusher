// FHFloat.02 — CLOSED round-2 execution primitives
// -----------------------------------------------------------------------------

// Purpose: flop has ended; round-2 action history is now final.
// Source/Provenance: T betround numbering.
##f$cc_hist_flop_float_closed##
IsTurn || IsRiver

// Purpose: prove Hero really executed the initial Float bet.
// Source: pre-action opportunity marker + closed OpenHoldem action counters.
// Provenance: A/T final-action proof.
// Safety: any prior check means a later check-raise/bet is not initial Float.
##f$cc_hist_flop_float_initial_bet_executed##
f$cc_hist_flop_float_closed &&
user_cc_flop_float_opportunity_seen &&
didchecround2 = 0 &&
(didbetsizeround2 > 0 || didalliround2 > 0)

// Purpose: initial Float was a normal sized bet.
// Source/Provenance: T round-2 betsize counter.
##f$cc_hist_flop_float_initial_bet_was_sized##
f$cc_hist_flop_float_initial_bet_executed && didbetsizeround2 > 0

// Purpose: initial Float itself was direct all-in.
// Source/Provenance: T counters + P initial-action disambiguation.
##f$cc_hist_flop_float_initial_bet_was_direct_allin##
f$cc_hist_flop_float_initial_bet_executed &&
didalliround2 > 0 &&
didbetsizeround2 = 0 &&
didraisround2 = 0

// Purpose: Hero declined available Float and checked back as exact LAST actor.
// Source: canonical opportunity is LAST, so this check closes the flop.
// Provenance: T/A exact check-back history.
##f$cc_hist_flop_float_checked_back##
f$cc_hist_flop_float_closed &&
user_cc_flop_float_opportunity_seen &&
didchecround2 > 0 &&
didcallround2 = 0 &&
didraisround2 = 0 &&
didbetsizeround2 = 0 &&
didalliround2 = 0

// Purpose: Float was later raised and Hero continued by calling.
// Source/Provenance: T action counters + final aggressor signal.
##f$cc_hist_flop_float_then_called_raise##
f$cc_hist_flop_float_initial_bet_executed &&
didcallround2 > 0 &&
lastraised2 != userchair

// Purpose: Hero Float-bet then re-aggressed on the flop.
// Source/Provenance: T multiple aggressive action counters.
##f$cc_hist_flop_float_then_reaggressed##
f$cc_hist_flop_float_initial_bet_executed &&
(didbetsizeround2 >= 2 || didraisround2 > 0 || (didbetsizeround2 > 0 && didalliround2 > 0))

// Purpose: Hero remains final flop aggressor after a normal one-size Float.
// Source/Provenance: T `lastraised2` + exact counter pattern.
##f$cc_hist_flop_float_standard_parent##
f$cc_hist_flop_float_closed &&
user_cc_flop_float_opportunity_seen &&
didchecround2 = 0 &&
didcallround2 = 0 &&
didraisround2 = 0 &&
didbetsizeround2 = 1 &&
didalliround2 = 0 &&
lastraised2 = userchair
