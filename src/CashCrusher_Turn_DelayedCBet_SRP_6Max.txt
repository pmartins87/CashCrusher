// Source boundary:
//   Crusher/DeepCrusher has literal two/three-handed delayed trees but no 4-6h
//   range tree for these exact cash-game states. Therefore no Spin frequency is
//   copied. The source contributes only A-level architecture: current-strength
//   reclassification, runout caution, IP/OOP and HU/multiway separation, and a
//   protected checking range. The action policy below is P-heavy professional
//   cash adaptation and is labelled as such.
//
// Safety:
//   * exact ordinary-SRP/PFA chronology is mandatory;
//   * HU and current-multiway are separate owners;
//   * pure-air Turn barrels are deliberately absent until a 4-6h flop-check
//     snapshot can prove a strategically selected bluff parent;
//   * no HandPower/random/BetMax/stackoff/commitment shortcut exists here;
//   * a Turn bet never authorizes a response versus a later raise.

// -----------------------------------------------------------------------------
// DCSR.01 — exact 4-6h ordinary-SRP ownership
// -----------------------------------------------------------------------------

// Purpose: source-silent 4-6h ordinary one-raise PFA parent after CLOSED flop X/X.
// Source/Provenance: T exact Gate00 chronology + Gate12A history; P strategy domain.
##f$cc_turn_delayed_cbet_srp6_parent##
f$cc_turn_delayed_cbet_base_opportunity &&
f$cc_deal_size >= 4 && f$cc_deal_size <= 6 &&
f$cc_pot_family_id = 2 &&
f$cc_pf_role_pfa &&
f$cc_pf_one_raise_ordinary_srp &&
f$cc_pf_single_raiser_pos_id = f$cc_hero_pos_id

// Purpose: 4-6h deal reduced preflop to exactly one caller; no postflop reduction
// is possible because the entire flop checked through.
// Source/Provenance: T exact field/history geometry.
##f$cc_turn_delayed_cbet_srp6_hu_context##
f$cc_turn_delayed_cbet_srp6_parent &&
f$cc_flop_entry_count = 2 &&
f$cc_hu &&
f$cc_hu_origin_preflop_reduced

// Purpose: ordinary-SRP PFA IP versus a surviving blind caller.
// Source/Provenance: T positional geometry + P exact range-family owner.
##f$cc_turn_delayed_cbet_srp6_hu_ip_blind##
f$cc_turn_delayed_cbet_srp6_hu_context &&
f$cc_hu_ip &&
(f$cc_hu_villain_pos_id = 5 || f$cc_hu_villain_pos_id = 6)

// Purpose: ordinary-SRP PFA OOP versus later-position nonblind caller.
// Source/Provenance: T positional geometry + P distinct cash range family.
##f$cc_turn_delayed_cbet_srp6_hu_oop_nonblind##
f$cc_turn_delayed_cbet_srp6_hu_context &&
f$cc_hu_oop &&
(f$cc_hu_villain_pos_id >= 1 && f$cc_hu_villain_pos_id <= 4)

// Purpose: exact reviewed HU family; weird reversed blind/nonblind topologies stay
// outside instead of inheriting a generic HU fallback.
// Source/Provenance: P deterministic partition over T geometry.
##f$cc_turn_delayed_cbet_srp6_hu_supported##
f$cc_turn_delayed_cbet_srp6_hu_ip_blind ||
f$cc_turn_delayed_cbet_srp6_hu_oop_nonblind

// Purpose: 3+ players remain after the checked-through flop.
// Source/Provenance: T field geometry + P multiway owner.
##f$cc_turn_delayed_cbet_srp6_mw_context##
f$cc_turn_delayed_cbet_srp6_parent &&
f$cc_flop_entry_count >= 3 &&
f$cc_multiway

// -----------------------------------------------------------------------------
// DCSR.02 — current strength / runout vocabulary
// -----------------------------------------------------------------------------

// Purpose: exact contributed current two pair below stronger value.
// Source/Provenance: T/A contribution-aware hand reconstruction.

// Purpose: contribution-aware value stronger than exact two pair.
// Source/Provenance: A portable value ladder; P policy remains downstream.
##f$cc_turn_delayed_cbet_srp6_strong_value##
f$cc_turn_delayed_cbet_two_pair_plus_real &&
!f$cc_turn_delayed_cbet_srp6_exact_two_pair

// Purpose: strong top-pair quality using audited kicker reconstruction.
// Source: Crusher repeatedly separates TP by kicker; exact <=2 boundary is P.
// Provenance: A source concept + P deterministic cash quality split.
##f$cc_turn_delayed_cbet_srp6_strong_tp##
f$cc_turn_delayed_cbet_top_pair_real && f$cc_number_better_kickers <= 2

// Purpose: medium top-pair band below strong TP.
// Source/Provenance: A/P deterministic quality split.
##f$cc_turn_delayed_cbet_srp6_medium_tp##
f$cc_turn_delayed_cbet_top_pair_real &&
f$cc_number_better_kickers > 2 && f$cc_number_better_kickers <= 4

// Purpose: weak top-pair residue.
// Source/Provenance: A/P explicit check-range descriptor.
##f$cc_turn_delayed_cbet_srp6_weak_tp##
f$cc_turn_delayed_cbet_top_pair_real && f$cc_number_better_kickers > 4

// Purpose: completion/four-card topology where one-pair and thin 2P value require
// more caution after the flop checked through.
// Source: direct Crusher delayed trees repeatedly treat completed boards specially.
// Provenance: A source principle + P 4-6h generalization.
##f$cc_turn_delayed_cbet_srp6_dangerous_runout##
f$cc_turn_completed ||
f$cc_turn_four_card_straight_or_flush ||
f$cc_turn_super_completed

// Purpose: clean Turn complement used by thin value/equity branches.
// Source/Provenance: P exact local complement.
##f$cc_turn_delayed_cbet_srp6_clean_turn##
!f$cc_turn_delayed_cbet_srp6_dangerous_runout

// Purpose: favorable A/K pressure overcard without a new completion.
// Source: source uses meaningful overcards as delayed-barrel ingredients.
// Provenance: A source concept + P cash selector; it does not authorize pure air.
##f$cc_turn_delayed_cbet_srp6_high_pressure_turn##
f$cc_turn_delayed_cbet_srp6_clean_turn &&
f$cc_turn_overcard_to_flop &&
!f$cc_turn_pairs_flop_rank &&
(f$cc_turn_is_ace || f$cc_turn_is_king)

// Purpose: premium current no-made draw without stack-off semantics.
// Source/Provenance: A cleaned Crusher draw quality + P exact combination boundary.
##f$cc_turn_delayed_cbet_srp6_premium_draw##
IsTurn && f$cc_hand_no_made &&
(f$cc_real_combo_draw || f$cc_real_nut_fd ||
 (f$cc_real_oesd && f$cc_two_overcards) ||
 (f$cc_real_fd && f$cc_two_overcards))

// Purpose: useful frontdoor draw below premium tier.
// Source/Provenance: A/P conservative semibluff descriptor.
##f$cc_turn_delayed_cbet_srp6_good_draw##
IsTurn && f$cc_hand_no_made &&
!f$cc_turn_delayed_cbet_srp6_premium_draw &&
(f$cc_real_fd || f$cc_real_oesd || (f$cc_real_gutshot && f$cc_two_overcards))
