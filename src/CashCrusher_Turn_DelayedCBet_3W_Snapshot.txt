// -----------------------------------------------------------------------------
// DC12A-3WS.00 — source-exact no-made draw / air vocabulary
// -----------------------------------------------------------------------------

// Purpose: source one-/two-hole flush-draw class. The second clause is equivalent
// to source Queen-high-or-better one-card FD (`NumberOfUnknownSuitedOvercards<=2`).
// Source/Provenance: T direct `f$hand_FD` mechanical translation.
##f$cc_turn_delayed_cbet_source_fd##
!IsRiver &&
((HaveFlushDraw && SuitsInHand = 1) ||
 (HaveFlushDraw && SuitsInHand = 2 && NumberOfUnknownSuitedOvercards <= 2))

// Purpose: source straight-draw class.
// Source/Provenance: T direct `f$hand_SD` translation.
##f$cc_turn_delayed_cbet_source_sd##
!IsRiver &&
((HaveStraightDraw && (nstraightfillcommon - nstraightfill = 2)) || HaveNutStraightDraw)

// Purpose: source gutshot class.
// Source/Provenance: T direct `f$hand_Gutshot` translation.
##f$cc_turn_delayed_cbet_source_gutshot##
!IsRiver &&
((HaveInsideStraightDraw && (nstraightfillcommon - nstraightfill = 2)) || HaveNutStraightDraw)

// Purpose: exact direct-frontdoor source draw union used by Retro12A.
// Source/Provenance: T `f$hand_Gutshot_or_BetterFDSD`.
##f$cc_turn_delayed_cbet_source_gutshot_or_better_fdsd##
f$cc_turn_delayed_cbet_source_sd ||
f$cc_turn_delayed_cbet_source_gutshot ||
f$cc_turn_delayed_cbet_source_fd

// Purpose: broader source draw-or-backdoor exclusion used to define `f$hand_Air`.
// Source also promotes two overcards + any backdoor and A/K-high + overcard +
// backdoor out of the air bucket on flop.
// Source/Provenance: T direct `f$hand_Gutshot_or_BetterDraw` translation.
##f$cc_turn_delayed_cbet_source_gutshot_or_better_draw##
When f$cc_turn_delayed_cbet_source_gutshot_or_better_fdsd Return true Force
When Overcards = 2 && (HaveBackdoorFlushDraw || HaveBackdoorStraightDraw) Return true Force
When IsFlop && (hand$A || hand$K) && Overcards >= 1 && (HaveBackdoorFlushDraw || HaveBackdoorStraightDraw) Return true Force
When Others Return false Force

// Purpose: direct source air for native BB/SB trees. The special BTNvSB
// stackoff-draw addendum in legacy `f$hand_Air` is intentionally not applied to
// BBvSB/SBvBB.
// Source/Provenance: T direct first clause of source `f$hand_Air`.
##f$cc_turn_delayed_cbet_source_air##
f$cc_hand_no_made && !f$cc_turn_delayed_cbet_source_gutshot_or_better_draw

// -----------------------------------------------------------------------------
// DC12A-3WS.01 — exact native HU-after-preflop-fold contexts
// -----------------------------------------------------------------------------

// Purpose: native 3-handed BTN initiative versus SB, with BB folded preflop.
// Source/Provenance: A exact `game_3wBTNvSB` reconstruction over T context.
##f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_context##
f$cc_turn_delayed_cbet_snapshot_capture_eligible &&
f$cc_deal_size = 3 &&
f$cc_flop_entry_count = 2 &&
f$cc_hu &&
f$cc_hero_pos_id = 4 &&
f$cc_hu_villain_pos_id = 5

// Purpose: native 3-handed BB initiative versus SB, with BTN folded preflop.
// Source/Provenance: A exact `game_3wBBvSB` reconstruction over T context.
##f$cc_turn_delayed_cbet_snapshot_3w_bbvsb_context##
f$cc_turn_delayed_cbet_snapshot_capture_eligible &&
f$cc_deal_size = 3 &&
f$cc_flop_entry_count = 2 &&
f$cc_hu &&
f$cc_hero_pos_id = 6 &&
f$cc_hu_villain_pos_id = 5

// -----------------------------------------------------------------------------
// DC12A-3WS.02 — exact 3wBTNvSB source candidates
// -----------------------------------------------------------------------------

// Purpose: source allows only flop 2nd/3rd pair to seed this delayed tree.
// Source/Provenance: T direct `FlopMPBPCheck` class via portable pair ladder.
##f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_mpbp_candidate##
f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_context &&
(f$cc_hand_second_pair_or_pocket || f$cc_hand_third_pair_or_pocket)

// Purpose: saved two-broadway texture for the direct delayed-50 condition.
// Source/Provenance: T direct `f$flop_2BW` translation.
##f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_2bw_candidate##
f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_mpbp_candidate && f$cc_flop_2bw

// Purpose: saved one-broadway texture for pair-9+ conditional delay.
// Source/Provenance: T direct `f$flop_1BW` translation.
##f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_1bw_candidate##
f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_mpbp_candidate && f$cc_flop_1bw

// Purpose: exact source pair-rank 9+ test. For a board-made pair tier the source
// uses Second/ThirdBestBoardCard; a pocket pair uses rankhiplayer.
// Source/Provenance: T direct Retro12A final-check condition.
##f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_pair9plus_candidate##
When !f$cc_turn_delayed_cbet_snapshot_3w_btnvsb_mpbp_candidate Return false Force
When f$cc_hand_second_pair_or_pocket && (SecondBestBoardCard >= 9 || (ispair && rankhiplayer >= 9)) Return true Force
When f$cc_hand_third_pair_or_pocket && (ThirdBestBoardCard >= 9 || (ispair && rankhiplayer >= 9)) Return true Force
When Others Return false Force

// -----------------------------------------------------------------------------
// DC12A-3WS.03 — 3wBBvSB source wetness reconstruction

// Purpose: exact source adjacency helper (KQ through 54; wheel edge deliberately
// follows the historical source rather than a generalized connectivity formula).
// Source/Provenance: T direct `f$TwoConnectedCardsOnFlop` translation.
##f$cc_turn_delayed_cbet_source_two_connected_flop##
((FirstFlopCard = 13 || SecondFlopCard = 13 || ThirdFlopCard = 13) && (FirstFlopCard = 12 || SecondFlopCard = 12 || ThirdFlopCard = 12)) ||
((FirstFlopCard = 12 || SecondFlopCard = 12 || ThirdFlopCard = 12) && (FirstFlopCard = 11 || SecondFlopCard = 11 || ThirdFlopCard = 11)) ||
((FirstFlopCard = 11 || SecondFlopCard = 11 || ThirdFlopCard = 11) && (FirstFlopCard = 10 || SecondFlopCard = 10 || ThirdFlopCard = 10)) ||
((FirstFlopCard = 10 || SecondFlopCard = 10 || ThirdFlopCard = 10) && (FirstFlopCard = 9 || SecondFlopCard = 9 || ThirdFlopCard = 9)) ||
((FirstFlopCard = 9 || SecondFlopCard = 9 || ThirdFlopCard = 9) && (FirstFlopCard = 8 || SecondFlopCard = 8 || ThirdFlopCard = 8)) ||
((FirstFlopCard = 8 || SecondFlopCard = 8 || ThirdFlopCard = 8) && (FirstFlopCard = 7 || SecondFlopCard = 7 || ThirdFlopCard = 7)) ||
((FirstFlopCard = 7 || SecondFlopCard = 7 || ThirdFlopCard = 7) && (FirstFlopCard = 6 || SecondFlopCard = 6 || ThirdFlopCard = 6)) ||
((FirstFlopCard = 6 || SecondFlopCard = 6 || ThirdFlopCard = 6) && (FirstFlopCard = 5 || SecondFlopCard = 5 || ThirdFlopCard = 5)) ||
((FirstFlopCard = 5 || SecondFlopCard = 5 || ThirdFlopCard = 5) && (FirstFlopCard = 4 || SecondFlopCard = 4 || ThirdFlopCard = 4))

// Purpose: source spread-out flop pattern: one J+, one 7-T, one 2-6.
// Source/Provenance: T direct `f$flop_spreaded` translation.
##f$cc_turn_delayed_cbet_source_flop_spreaded##
(FirstFlopCard >= 11 || SecondFlopCard >= 11 || ThirdFlopCard >= 11) &&
((FirstFlopCard >= 7 && FirstFlopCard <= 10) || (SecondFlopCard >= 7 && SecondFlopCard <= 10) || (ThirdFlopCard >= 7 && ThirdFlopCard <= 10)) &&
((FirstFlopCard >= 2 && FirstFlopCard <= 6) || (SecondFlopCard >= 2 && SecondFlopCard <= 6) || (ThirdFlopCard >= 2 && ThirdFlopCard <= 6))

// Purpose: source flop draw-heavy predicate.
// Source/Provenance: T direct `f$flop_DrawHeavy` translation.
##f$cc_turn_delayed_cbet_source_flop_drawheavy##
SuitsOnFlop <= 2 && OpenEndedStraightDrawPossibleOnFlop && !PairOnBoard

// Purpose: exact 3wBBvSB branch of legacy `f$board_dry` on flop. OpponentsAtTable
// is 2 in the native three-handed source, so the HU-table-only dry helper does
// not participate. Early draw-heavy false branches retain precedence.
// Source/Provenance: T direct source mechanics, A normalized flat expression.
##f$cc_turn_delayed_cbet_source_3w_bbvsb_flop_dry##
When FlushDrawPossible && OpenEndedStraightDrawPossibleOnFlop Return false Force
When f$cc_turn_delayed_cbet_source_flop_drawheavy Return false Force
When SuitsOnFlop >= 2 && !f$cc_turn_delayed_cbet_source_two_connected_flop && f$cc_turn_delayed_cbet_source_flop_spreaded Return true Force
When SuitsOnFlop >= 3 && !StraightPossibleOnFlop Return true Force
When SuitsOnFlop >= 3 && !OpenEndedStraightDrawPossibleOnFlop Return true Force
When Others Return false Force

// Purpose: direct source wet-flop complement used only inside native 3wBBvSB.
// Source/Provenance: T complement of the exact source dry predicate.
##f$cc_turn_delayed_cbet_source_3w_bbvsb_flop_wet##
IsFlop && !f$cc_turn_delayed_cbet_source_3w_bbvsb_flop_dry
