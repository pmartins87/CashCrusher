// Purpose: prove fraction*pot Turn-Float geometry is clean before execution.
// Source: canonical Turn Float is Hero's first turn action with no bet facing Hero.
// Provenance: T exact currentbet/potplayer/AmountToCall runtime guard.
##f$cc_turn_float_stackgeom_clean##
f$cc_turn_float_runtime_size_ready &&
f$cc_turn_float_runtime_mapping_valid &&
AmountToCall = 0 &&
currentbet = 0 &&
potplayer = 0 &&
f$cc_round_start_pot_bb > 0 &&
f$cc_hero_stack_round_start_bb > 0

// Purpose: requested Turn Float contribution in big blinds.
// Source/Provenance: T exact initial-bet percentage arithmetic.
##f$cc_turn_float_requested_bet_bb##
When !f$cc_turn_float_stackgeom_clean Return -1 Force
When f$cc_turn_float_requested_pot_fraction <= 0 Return -1 Force
When Others Return (f$cc_turn_float_requested_pot_fraction * f$cc_round_start_pot_bb) Force

// Purpose: requested bet / Hero available turn-start stack.
// Source/Provenance: T mechanical balance ratio.
##f$cc_turn_float_requested_hero_balance_ratio##
When !f$cc_turn_float_stackgeom_clean Return -1 Force
When f$cc_hero_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_float_requested_bet_bb / f$cc_hero_stack_round_start_bb) Force

// Purpose: requested bet / exact current HU effective stack.
// Source/Provenance: T/A HU effective geometry.
##f$cc_turn_float_requested_hu_effective_ratio##
When !f$cc_turn_float_stackgeom_clean Return -1 Force
When !f$cc_hu Return -1 Force
When f$cc_hu_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_float_requested_bet_bb / f$cc_hu_effective_stack_round_start_bb) Force

// Purpose: requested bet / shallowest multiway effective relation.
// Source/Provenance: T/A dual-bound multiway geometry.
// Safety: reaching only this denominator can create one short sidepot.
##f$cc_turn_float_requested_mw_shallowest_ratio##
When !f$cc_turn_float_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_shallowest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_float_requested_bet_bb / f$cc_mw_shallowest_effective_stack_round_start_bb) Force

// Purpose: requested bet / deepest all-live multiway effective relation.
// Source/Provenance: T/A whole-field denominator.
##f$cc_turn_float_requested_mw_deepest_ratio##
When !f$cc_turn_float_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_deepest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_float_requested_bet_bb / f$cc_mw_deepest_effective_stack_round_start_bb) Force

// Purpose: requested size already exhausts Hero stack.
// Source/Provenance: T mechanical reach.
##f$cc_turn_float_requested_reaches_hero_stack##
f$cc_turn_float_requested_hero_balance_ratio >= 1.00

// Purpose: requested size reaches exact HU effective relation.
// Source/Provenance: T/A HU mechanical reach.
##f$cc_turn_float_requested_reaches_hu_effective##
f$cc_hu && f$cc_turn_float_requested_hu_effective_ratio >= 1.00

// Purpose: requested size reaches at least one shallow multiway relation.
// Source/Provenance: T/A sidepot descriptor only.
##f$cc_turn_float_requested_reaches_mw_shallowest##
f$cc_multiway && f$cc_turn_float_requested_mw_shallowest_ratio >= 1.00

// Purpose: requested size reaches deepest/all-live multiway relation.
// Source/Provenance: T/A whole-field mechanical reach.
##f$cc_turn_float_requested_reaches_mw_deepest##
f$cc_multiway && f$cc_turn_float_requested_mw_deepest_ratio >= 1.00

// Purpose: short sidepot all-in while deeper opponent remains matchable.
// Source/Provenance: P/T sidepot safety diagnostic.
##f$cc_turn_float_mw_sidepot_divergence_candidate##
f$cc_turn_float_requested_reaches_mw_shallowest && !f$cc_turn_float_requested_reaches_mw_deepest

// Purpose: preserve historical near-all-in threshold as a review signal only.
// Source: DeepCrusher frequently promoted bets around 50/55/60% of balance/effective.
// Provenance: A diagnostic, never a generic cash shove instruction.
##f$cc_turn_float_historical_near_allin_review##
When !f$cc_turn_float_stackgeom_clean Return false Force
When f$cc_turn_float_requested_hero_balance_ratio >= 0.50 Return true Force
When f$cc_hu && f$cc_turn_float_requested_hu_effective_ratio >= 0.50 Return true Force
When f$cc_multiway && f$cc_turn_float_requested_mw_deepest_ratio >= 0.50 Return true Force
When Others Return false Force
