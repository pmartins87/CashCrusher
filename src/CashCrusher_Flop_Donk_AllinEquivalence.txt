// Purpose:
//   Convert a reviewed 50/75/POT Donk to BetMax only when the requested size
//   already reaches Hero's stack or every currently live effective relation.
//
// Source boundary:
//   The source's draw-heavy low-SPR POT request remains POT unless mechanically
//   all-in-equivalent. Historical ~50/55/60 thresholds and future Turn-jam plans
//   are not generic Flop Donk promotions here.
//
// Source/Provenance:
//   T = OpenPPL BetMax and exact stack geometry.
//   A = mechanical-equivalence separation frozen in prior attack gates.
//   P = multiway sidepot safety and pre-action plan metadata.
// ============================================================================

// Purpose: requested Donk covers every currently matchable opponent relation.
// Source/Provenance: T/A HU exact or multiway deepest-effective definition.
##f$cc_flop_donk_requested_reaches_all_live_effective##
When !f$cc_flop_donk_stackgeom_clean Return false Force
When f$cc_hu && f$cc_flop_donk_requested_reaches_hu_effective Return true Force
When f$cc_multiway && f$cc_flop_donk_requested_reaches_mw_deepest Return true Force
When Others Return false Force

// Purpose: natural/mechanical local BetMax equivalence only.
// Source/Provenance: T/A mechanical cap; not strategic commitment.
##f$cc_flop_donk_natural_allin_equivalent##
When !f$cc_flop_donk_runtime_size_ready Return false Force
When !f$cc_flop_donk_stackgeom_clean Return false Force
When f$cc_flop_donk_requested_reaches_hero_stack Return true Force
When f$cc_flop_donk_requested_reaches_all_live_effective Return true Force
When Others Return false Force

// Purpose: reason ID: 1 Hero stack | 2 HU effective | 3 MW deepest/all-live.
// Source/Provenance: P metadata over T/A mechanical conditions.
##f$cc_flop_donk_natural_allin_reason_id##
When !f$cc_flop_donk_natural_allin_equivalent Return 0 Force
When f$cc_flop_donk_requested_reaches_hero_stack Return 1 Force
When f$cc_hu && f$cc_flop_donk_requested_reaches_hu_effective Return 2 Force
When f$cc_multiway && f$cc_flop_donk_requested_reaches_mw_deepest Return 3 Force
When Others Return 0 Force

// Purpose: final local Donk execution size plus PRE-ACTION plan markers.
// Source/Provenance: T/A native sizing/BetMax adapter + P history metadata.
// Safety: user markers describe the plan and never prove execution.
##f$cc_flop_donk_execution_betsize##
When !f$cc_flop_donk_runtime_size_ready Return 0 Force
When f$cc_flop_donk_runtime_size_ready Set user_cc_flop_donk_plan_bet_seen
When f$cc_flop_donk_runtime_size_ready Set user_cc_flop_donk_plan_size_recorded
When f$cc_flop_donk_size_id = 1 Set user_cc_flop_donk_plan_size_50
When f$cc_flop_donk_size_id = 2 Set user_cc_flop_donk_plan_size_75
When f$cc_flop_donk_size_id = 3 Set user_cc_flop_donk_plan_size_100
When f$cc_flop_donk_natural_allin_equivalent Set user_cc_flop_donk_plan_expected_allin
When f$cc_flop_donk_natural_allin_equivalent Return BetMax Force
When Others Return f$cc_flop_donk_requested_betsize Force

// Purpose: count mutually-exclusive Donk plan-size markers.
// Source/Provenance: P integrity check over per-hand booleans.
##f$cc_flop_donk_plan_size_marker_count##
(user_cc_flop_donk_plan_size_50 * 1) +
(user_cc_flop_donk_plan_size_75 * 1) +
(user_cc_flop_donk_plan_size_100 * 1)

// Purpose: validate stored Donk plan markers without recomputing old flop stack.
// Source/Provenance: P cross-street-stable marker invariant.
##f$cc_flop_donk_plan_markers_consistent##
When user_cc_flop_donk_plan_size_recorded && !user_cc_flop_donk_plan_bet_seen Return false Force
When user_cc_flop_donk_plan_bet_seen && !user_cc_flop_donk_plan_size_recorded Return false Force
When user_cc_flop_donk_plan_size_recorded && f$cc_flop_donk_plan_size_marker_count != 1 Return false Force
When !user_cc_flop_donk_plan_size_recorded && f$cc_flop_donk_plan_size_marker_count != 0 Return false Force
When user_cc_flop_donk_plan_expected_allin && !user_cc_flop_donk_plan_size_recorded Return false Force
When Others Return true Force

// Purpose: shortest-only sidepot reach must not promote whole-field Donk to BetMax.
// Source/Provenance: P invariant over T dual-bound geometry.
##f$cc_flop_donk_sidepot_divergence_not_promoted##
When !f$cc_flop_donk_mw_sidepot_divergence_candidate Return true Force
When f$cc_flop_donk_natural_allin_equivalent Return false Force
When Others Return true Force
