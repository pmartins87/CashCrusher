// -----------------------------------------------------------------------------
// RG.00 — Common P-heavy descriptors
// -----------------------------------------------------------------------------

// Purpose: clean enough River for one-pair thin/value pressure.
// Source/Provenance: A runout caution + P six-max abstraction.
##f$cc_river_srp_gap_clean##
IsRiver && !f$cc_river_completed && !f$cc_river_four_card_completion

// Purpose: clean A/K or otherwise meaningful overcard pressure River.
// Source/Provenance: P range-advantage/bluff selector, never sufficient alone.
##f$cc_river_srp_gap_high_pressure##
f$cc_river_srp_gap_clean &&
f$cc_river_meaningful_overcard &&
!f$cc_river_pairs_previous_board

// Purpose: prior Turn size selected the caller materially (75% or pot).
// Source/Provenance: P range-selection modifier over reviewed Turn plan.
##f$cc_river_srp_gap_large_turn_barrel##
f$cc_river_cbet_opportunity && f$cc_river_turn_plan_size_id >= 6

// Purpose: all current River multiway effective relationships have SPR <2.
// Source/Provenance: T/P deepest-effective geometry fact only.
##f$cc_river_mw_all_effective_spr_below_2##
f$cc_river_mw_deep_spr_valid && f$cc_mw_spr_deepest_round_start < 2

// Purpose: all current River multiway effective relationships have SPR <1.
// Source/Provenance: T/P geometry fact only; no automatic jam semantics.
##f$cc_river_mw_all_effective_spr_below_1##
f$cc_river_mw_deep_spr_valid && f$cc_mw_spr_deepest_round_start < 1

// -----------------------------------------------------------------------------
// RG.01 — UTG/HJ/CO PFA IP versus BB
// -----------------------------------------------------------------------------

// Purpose: P-heavy River after Gate02 family2 IP-v-BB second barrel.
// Source: no literal DeepCrusher six-max range family.
// Provenance: P professional-theory fill over exact Gate02 range/history.
##f$cc_river_srp_gap_ip_vs_bb_context##
f$cc_river_turn_hu_still_hu &&
f$cc_river_turn_family_id = 2 &&
f$cc_hu_ip &&
f$cc_hu_villain_pos_id = 6

// Purpose: IP-v-BB third-barrel action.
// Professional fill: BB continues widest, so clean OP/strong TP retain value;
// medium TP thin-values only after non-large Turn barrel at manageable SPR;
// missed Turn draw bluff is narrowly selected by clean pressure + A/K blocker.
// Provenance: P; no solver-exact frequency claim.
##f$cc_river_srp_gap_ip_vs_bb_action##
When !f$cc_river_srp_gap_ip_vs_bb_context Return false Force
When f$cc_river_two_pair_plus Return true Force
When f$cc_river_overpair && f$cc_river_srp_gap_clean Return true Force
When f$cc_river_overpair && f$cc_river_hu_spr_below_1 Return true Force
When f$cc_river_overpair Return false Force
When f$cc_river_strong_top_pair && f$cc_river_srp_gap_clean Return true Force
When f$cc_river_medium_top_pair && f$cc_river_srp_gap_clean && !f$cc_river_srp_gap_large_turn_barrel && !f$cc_river_hu_high_spr Return true Force
When f$cc_river_top_pair Return false Force
When f$cc_river_second_pair Return false Force
When f$cc_river_no_made && f$cc_river_turn_had_draw && f$cc_river_srp_gap_high_pressure && !f$cc_river_srp_gap_large_turn_barrel && (f$cc_hole_has_ace || f$cc_hole_has_king) Return true Force
When f$cc_river_no_made Return false Force
When Others Return false Force

// Purpose: IP-v-BB P-heavy River size map.
// Source/Provenance: P: 75 robust/polar, 50 one-pair value, 33 medium thin value.
##f$cc_river_srp_gap_ip_vs_bb_size_id##
When !f$cc_river_srp_gap_ip_vs_bb_action Return 0 Force
When f$cc_river_two_pair_plus Return f$cc_river_size_75_id Force
When f$cc_river_overpair Return f$cc_river_size_50_id Force
When f$cc_river_strong_top_pair Return f$cc_river_size_50_id Force
When f$cc_river_medium_top_pair Return f$cc_river_size_33_id Force
When f$cc_river_no_made Return f$cc_river_size_75_id Force
When Others Return 0 Force
