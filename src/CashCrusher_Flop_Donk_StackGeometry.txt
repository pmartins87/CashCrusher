// Purpose: prove simple fraction*pot first-action Donk geometry is clean.
// Source: canonical Donk opportunity is Hero first flop action with no bet faced.
// Provenance: T exact currentbet/potplayer/AmountToCall guard.
##f$cc_flop_donk_stackgeom_clean##
f$cc_flop_donk_runtime_size_ready &&
f$cc_flop_donk_runtime_mapping_valid &&
AmountToCall = 0 &&
currentbet = 0 &&
potplayer = 0 &&
f$cc_round_start_pot_bb > 0 &&
f$cc_hero_stack_round_start_bb > 0

// Purpose: requested Donk contribution in big blinds.
// Source/Provenance: T exact initial-bet arithmetic.
##f$cc_flop_donk_requested_bet_bb##
When !f$cc_flop_donk_stackgeom_clean Return -1 Force
When f$cc_flop_donk_requested_pot_fraction <= 0 Return -1 Force
When Others Return (f$cc_flop_donk_requested_pot_fraction * f$cc_round_start_pot_bb) Force

// Purpose: requested Donk / Hero available flop-start stack.
// Source/Provenance: T mechanical balance ratio.
##f$cc_flop_donk_requested_hero_balance_ratio##
When !f$cc_flop_donk_stackgeom_clean Return -1 Force
When f$cc_hero_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_flop_donk_requested_bet_bb / f$cc_hero_stack_round_start_bb) Force

// Purpose: requested Donk / exact HU effective stack.
// Source/Provenance: T/A HU effective geometry.
##f$cc_flop_donk_requested_hu_effective_ratio##
When !f$cc_flop_donk_stackgeom_clean Return -1 Force
When !f$cc_hu Return -1 Force
When f$cc_hu_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_flop_donk_requested_bet_bb / f$cc_hu_effective_stack_round_start_bb) Force

// Purpose: requested Donk / shallowest multiway effective relation.
// Source/Provenance: T/A dual-bound geometry; sidepot descriptor only.
##f$cc_flop_donk_requested_mw_shallowest_ratio##
When !f$cc_flop_donk_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_shallowest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_flop_donk_requested_bet_bb / f$cc_mw_shallowest_effective_stack_round_start_bb) Force

// Purpose: requested Donk / deepest all-live multiway effective relation.
// Source/Provenance: T/A whole-field effective denominator.
##f$cc_flop_donk_requested_mw_deepest_ratio##
When !f$cc_flop_donk_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_deepest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_flop_donk_requested_bet_bb / f$cc_mw_deepest_effective_stack_round_start_bb) Force

// Purpose: reviewed Donk already exhausts Hero stack.
// Source/Provenance: T mechanical reach.
##f$cc_flop_donk_requested_reaches_hero_stack##
f$cc_flop_donk_requested_hero_balance_ratio >= 1.00

// Purpose: reviewed Donk reaches exact HU effective relation.
// Source/Provenance: T/A HU mechanical reach.
##f$cc_flop_donk_requested_reaches_hu_effective##
f$cc_hu && f$cc_flop_donk_requested_hu_effective_ratio >= 1.00

// Purpose: reviewed Donk reaches at least one shallow MW relationship.
// Source/Provenance: T/A sidepot diagnostic only.
##f$cc_flop_donk_requested_reaches_mw_shallowest##
f$cc_multiway && f$cc_flop_donk_requested_mw_shallowest_ratio >= 1.00

// Purpose: reviewed Donk reaches deepest/all-live MW relationship.
// Source/Provenance: T/A whole-field mechanical reach.
##f$cc_flop_donk_requested_reaches_mw_deepest##
f$cc_multiway && f$cc_flop_donk_requested_mw_deepest_ratio >= 1.00

// Purpose: detect shortest-only sidepot all-in while deeper players remain.
// Source/Provenance: P/T sidepot safety diagnostic.
##f$cc_flop_donk_mw_sidepot_divergence_candidate##
f$cc_flop_donk_requested_reaches_mw_shallowest && !f$cc_flop_donk_requested_reaches_mw_deepest

// Purpose: historical short-stack near-all-in signal retained for future exact
// node review, never a generic action.
// Source: DeepCrusher ~50/55/60 commitment/sizing evidence.
// Provenance: A diagnostic after the user's binding review-not-delete rule.
##f$cc_flop_donk_historical_near_allin_review##
When !f$cc_flop_donk_stackgeom_clean Return false Force
When f$cc_flop_donk_requested_hero_balance_ratio >= 0.50 Return true Force
When f$cc_hu && f$cc_flop_donk_requested_hu_effective_ratio >= 0.50 Return true Force
When f$cc_multiway && f$cc_flop_donk_requested_mw_deepest_ratio >= 0.50 Return true Force
When Others Return false Force
