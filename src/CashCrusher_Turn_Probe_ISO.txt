// Purpose: canonical Hero-position bit for preflop caller-mask intersections.
// Source/Provenance: T/A Gate00 canonical position mapping.
##f$cc_turn_probe_iso_hero_pos_bit##
When f$cc_hero_pos_id = 1 Return 1 Force
When f$cc_hero_pos_id = 2 Return 2 Force
When f$cc_hero_pos_id = 3 Return 4 Force
When f$cc_hero_pos_id = 4 Return 8 Force
When f$cc_hero_pos_id = 5 Return 16 Force
When f$cc_hero_pos_id = 6 Return 32 Force
When Others Return 0 Force

// Purpose: Hero limped before the sole isolation raise.
// Source/Provenance: A exact intersection with Gate01E limper mask.
##f$cc_turn_probe_iso_hero_was_limper##
f$cc_pf_iso_proven &&
f$cc_turn_probe_iso_hero_pos_bit > 0 &&
((f$cc_pf_pre_raise_limper_mask BitAnd f$cc_turn_probe_iso_hero_pos_bit) > 0)

// Purpose: Hero entered only after the isolation raise.
// Source/Provenance: A exact intersection with Gate01E post-raise mask.
##f$cc_turn_probe_iso_hero_was_postraise_coldcaller##
f$cc_pf_iso_proven &&
f$cc_turn_probe_iso_hero_pos_bit > 0 &&
((f$cc_pf_post_raise_coldcaller_mask BitAnd f$cc_turn_probe_iso_hero_pos_bit) > 0)

// Purpose: caller stage must be singular and proven.
// Source/Provenance: P fail-closed invariant over A exact masks.
##f$cc_turn_probe_iso_hero_origin_consistent##
When !f$cc_pf_iso_proven Return false Force
When f$cc_turn_probe_iso_hero_was_limper && f$cc_turn_probe_iso_hero_was_postraise_coldcaller Return false Force
When f$cc_turn_probe_iso_hero_was_limper || f$cc_turn_probe_iso_hero_was_postraise_coldcaller Return true Force
When Others Return false Force

// Purpose: canonical ISO Probe context.
// Source/Provenance: T/A exact history owner + P caller-stage requirement.
##f$cc_turn_probe_iso_context##
f$cc_turn_probe_base_opportunity &&
f$cc_turn_probe_snapshot_valid &&
f$cc_pot_family_id = 2 &&
f$cc_pf_iso_proven &&

f$cc_turn_probe_iso_hero_origin_consistent

// Purpose: conservative four-card bluff-pressure veto.
// Source: paired/A/2+BW pressure principle from source Probe material.
// Provenance: A source concept + P completed-board cash caution.
##f$cc_turn_probe_iso_bad_bluff_turn##
PairOnBoard ||
AcePresentOnFlop ||
f$cc_turn_is_ace ||
user_cc_turn_probe_flop_had_2plusbw ||
(user_cc_turn_probe_flop_had_1bw && f$cc_turn_is_broadway) ||
f$cc_turn_completed

// Purpose: strong exact TP for value/protection probes.
// Source/Provenance: A DeepCrusher kicker stratification + P <=2 split.
##f$cc_turn_probe_iso_strong_tp##
f$cc_hand_just_top_pair && f$cc_number_better_kickers <= 2

// Purpose: premium no-made draw, no stackoff implication.
// Source/Provenance: A/P cleaned high-equity draw family.
##f$cc_turn_probe_iso_premium_draw##
f$cc_hand_no_made &&
(f$cc_real_combo_draw || f$cc_real_nut_fd ||
 (f$cc_real_oesd && f$cc_two_overcards) ||
 (f$cc_real_fd && f$cc_two_overcards))

// Purpose: late isolator band where original-limper air pressure is least unsafe.
// Source/Provenance: P six-max range-strength grouping; exact raiser ID retained.

f$cc_pf_single_raiser_pos_id = 5 ||
f$cc_pf_single_raiser_pos_id = 6

// Purpose: narrow original-limper selected air candidate.
// Source: general Probe pressure principle after missed flop aggression.
// Provenance: P-heavy; requires late isolator + saved backdoor + favorable Turn.
