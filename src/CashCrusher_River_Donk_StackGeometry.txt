// ============================================================================
// CashCrusher_River_Donk_StackGeometry.txt
// Gate 09H.2 — mechanical River-Donk stack-sensitive execution geometry
// ============================================================================
// Purpose:
//   Measure the already-selected River-Donk size against Hero balance and exact
//   HU / dual-bound multiway effective stacks. Historical Spin near-all-in
//   thresholds are retained only as diagnostics, never as automatic action.
//
// Source/Provenance:
//   T = OpenHoldem pot/balance/currentbet + Gate00 effective-stack geometry.
//   A = historical ~50/60 review thresholds retained for audit visibility.
//   P = dual-bound sidepot safety and whole-field equivalence rules.
//
// Safety:
//   * canonical River Donk is an initial first-action bet;
//   * shortest-only multiway reach is a sidepot event, not whole-field all-in;
//   * deepest effective relation controls all-live equivalence;
//   * no BetMax is returned by this file.
// ============================================================================

// Purpose: requested River Donk is a clean initial street bet whose simple
// fraction*street-start-pot arithmetic is valid.
// Source/Provenance: T runtime current-bet/pot facts + P ownership guard.
##f$cc_river_donk_stackgeom_clean##
f$cc_river_donk_runtime_size_ready &&
f$cc_river_donk_runtime_mapping_valid &&
AmountToCall = 0 &&
currentbet = 0 &&
potplayer = 0 &&
f$cc_round_start_pot_bb > 0 &&
f$cc_hero_stack_round_start_bb > 0

// Purpose: requested River Donk contribution in BB.
// Source/Provenance: T/A exact initial-bet arithmetic.
##f$cc_river_donk_requested_bet_bb##
When !f$cc_river_donk_stackgeom_clean Return -1 Force
When f$cc_river_donk_requested_pot_fraction <= 0 Return -1 Force
When Others Return (f$cc_river_donk_requested_pot_fraction * f$cc_round_start_pot_bb) Force

// Purpose: Hero residual stack after requested River Donk.
// Source/Provenance: T arithmetic; negative means requested size exceeds balance.
##f$cc_river_donk_hero_residual_after_bet_bb##
When f$cc_river_donk_requested_bet_bb < 0 Return -1 Force
When Others Return (f$cc_hero_stack_round_start_bb - f$cc_river_donk_requested_bet_bb) Force

// Purpose: requested Donk / Hero street-start stack.
// Source/Provenance: T exact geometry.
##f$cc_river_donk_requested_hero_balance_ratio##
When !f$cc_river_donk_stackgeom_clean Return -1 Force
When f$cc_hero_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_river_donk_requested_bet_bb / f$cc_hero_stack_round_start_bb) Force

// Purpose: historical ~50%-Hero-stack review signal only.
// Source/Provenance: A DeepCrusher threshold evidence, never action.
##f$cc_river_donk_hist_hero_balance_50_trigger##
f$cc_river_donk_requested_hero_balance_ratio >= 0.50

// Purpose: historical ~60%-Hero-stack review signal only.
// Source/Provenance: A sizing-promotion ancestry, never action.
##f$cc_river_donk_hist_hero_stack_60_trigger##
f$cc_river_donk_requested_hero_balance_ratio >= 0.60

// Purpose: requested River Donk / exact HU effective stack.
// Source/Provenance: T/A exact HU relationship.
##f$cc_river_donk_requested_hu_effective_ratio##
When !f$cc_river_donk_stackgeom_clean Return -1 Force
When !f$cc_hu Return -1 Force
When f$cc_hu_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_river_donk_requested_bet_bb / f$cc_hu_effective_stack_round_start_bb) Force

// Purpose: historical ~50%-HU-effective review signal only.
// Source/Provenance: A callback ancestry, no action.
##f$cc_river_donk_hist_hu_effective_50_trigger##
f$cc_river_donk_requested_hu_effective_ratio >= 0.50

// Purpose: requested Donk / shallowest multiway effective relation.
// Source/Provenance: T/A dual-bound sidepot descriptor.
##f$cc_river_donk_requested_mw_shallowest_effective_ratio##
When !f$cc_river_donk_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_shallowest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_river_donk_requested_bet_bb / f$cc_mw_shallowest_effective_stack_round_start_bb) Force

// Purpose: requested Donk / deepest current multiway effective relation.
// Source/Provenance: T/A whole-field denominator.
##f$cc_river_donk_requested_mw_deepest_effective_ratio##
When !f$cc_river_donk_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_deepest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_river_donk_requested_bet_bb / f$cc_mw_deepest_effective_stack_round_start_bb) Force

// Purpose: historical ~50%-effective multiway review signal using deepest/all-live.
// Source/Provenance: A migration diagnostic only.
##f$cc_river_donk_hist_mw_effective_50_trigger##
f$cc_river_donk_requested_mw_deepest_effective_ratio >= 0.50

// Purpose: requested size reaches Hero's entire available stack.
// Source/Provenance: T mechanical balance geometry.
##f$cc_river_donk_requested_reaches_hero_stack##
f$cc_river_donk_requested_hero_balance_ratio >= 1.00

// Purpose: requested size reaches exact HU effective relationship.
// Source/Provenance: T exact HU arithmetic.
##f$cc_river_donk_requested_reaches_hu_effective##
f$cc_hu && f$cc_river_donk_requested_hu_effective_ratio >= 1.00

// Purpose: requested size reaches shortest MW effective relation.
// Source/Provenance: T/A sidepot descriptor only.
##f$cc_river_donk_requested_reaches_mw_shallowest##
f$cc_multiway && f$cc_river_donk_requested_mw_shallowest_effective_ratio >= 1.00

// Purpose: requested size reaches deepest/all-live MW effective relation.
// Source/Provenance: T/A whole-field descriptor.
##f$cc_river_donk_requested_reaches_mw_deepest##
f$cc_multiway && f$cc_river_donk_requested_mw_deepest_effective_ratio >= 1.00

// Purpose: shortest-only sidepot reach while deeper Villain remains.
// Source/Provenance: P/T diagnostic; never whole-field BetMax.
##f$cc_river_donk_mw_sidepot_divergence_candidate##
f$cc_river_donk_requested_reaches_mw_shallowest &&
!f$cc_river_donk_requested_reaches_mw_deepest

// Purpose: stack-sensitive geometry engaged and deserves audit visibility.
// True means review, never automatic shove.
// Source/Provenance: A/P metadata over T geometry.
##f$cc_river_donk_stack_sensitive_review_required##
When !f$cc_river_donk_stackgeom_clean Return false Force
When f$cc_river_donk_hist_hero_balance_50_trigger Return true Force
When f$cc_hu && f$cc_river_donk_hist_hu_effective_50_trigger Return true Force
When f$cc_multiway && f$cc_river_donk_hist_mw_effective_50_trigger Return true Force
When f$cc_river_donk_requested_reaches_hero_stack Return true Force
When f$cc_hu && f$cc_river_donk_requested_reaches_hu_effective Return true Force
When f$cc_multiway && f$cc_river_donk_requested_reaches_mw_shallowest Return true Force
When Others Return false Force

// Purpose: compact diagnostic bit flags.
// Bits: 1 Hero>=50; 2 Hero>=60; 4 HU-eff>=50; 8 reaches HU effective;
// 16 reaches MW shallowest; 32 MW-deepest>=50; 64 reaches MW deepest;
// 128 shortest-only sidepot divergence.
// Source/Provenance: P metadata over A/T diagnostics.
##f$cc_river_donk_stack_review_flags_id##
When !f$cc_river_donk_stackgeom_clean Return 0 Force
When Others Return ((f$cc_river_donk_hist_hero_balance_50_trigger * 1) + (f$cc_river_donk_hist_hero_stack_60_trigger * 2) + ((f$cc_hu && f$cc_river_donk_hist_hu_effective_50_trigger) * 4) + ((f$cc_hu && f$cc_river_donk_requested_reaches_hu_effective) * 8) + ((f$cc_multiway && f$cc_river_donk_requested_reaches_mw_shallowest) * 16) + ((f$cc_multiway && f$cc_river_donk_hist_mw_effective_50_trigger) * 32) + ((f$cc_multiway && f$cc_river_donk_requested_reaches_mw_deepest) * 64) + ((f$cc_multiway && f$cc_river_donk_mw_sidepot_divergence_candidate) * 128)) Force
