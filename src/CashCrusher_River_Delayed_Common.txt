// Purpose: source completed River = straight or flush is possible on public board.
// Source/Provenance: T direct mature `f$river_Completed` translation.
##f$cc_river_delayed_completed##
IsRiver && (FlushPossible || StraightPossible)

// Purpose: contribution-aware current River 2P+ / nut tier. Board-only made hands
// do not silently become source value.
// Source/Provenance: T/A CF7-safe mechanical hierarchy.
##f$cc_river_delayed_two_pair_plus_real##
When !IsRiver Return false Force
When HaveNuts Return true Force
When HaveStraightFlush && pokerval > pokervalcommon Return true Force
When HaveQuads && pokerval > pokervalcommon Return true Force
When HaveFullHouse && pokerval > pokervalcommon Return true Force
When HaveFlush && npcbits > 0 && pokerval > pokervalcommon Return true Force
When HaveStraight && npcbits > 0 && pokerval > pokervalcommon Return true Force
When HaveTrips && !TripsOnBoard && npcbits > 0 Return true Force
When HaveTwoPair && !TwoPairOnBoard && npcbits > 0 Return true Force
When Others Return false Force

// Purpose: current real top pair below 2P+.
// Source/Provenance: T/A contribution-aware pair hierarchy.
##f$cc_river_delayed_top_pair_real##
IsRiver &&
f$cc_hand_just_top_pair &&
!TwoPairOnBoard &&
npcbits > 0 &&
!f$cc_river_delayed_two_pair_plus_real

// Purpose: current real overpair below 2P+.
// Source/Provenance: T/A contribution-aware pocket-pair hierarchy.
##f$cc_river_delayed_overpair_real##
IsRiver &&
HaveOverPair && PairInHand && npcbits > 0 &&
!f$cc_river_delayed_two_pair_plus_real

// Purpose: source TP/OP current-value union.
// Source/Provenance: A mechanical convenience union.
##f$cc_river_delayed_tpop_real##
f$cc_river_delayed_top_pair_real || f$cc_river_delayed_overpair_real

// Purpose: current second-pair/pocket tier below TP.
// Source/Provenance: T/A portable pair ladder.
##f$cc_river_delayed_second_pair##
IsRiver &&
!f$cc_hand_top_pair_or_better &&
f$cc_hand_second_pair_or_pocket

// Purpose: exact current third-pair/pocket tier used by HUBB source MIN rule.
// Source/Provenance: T/A portable pair ladder; fourth/worse deliberately excluded.
##f$cc_river_delayed_third_pair##
IsRiver &&
!f$cc_hand_top_pair_or_better &&
!f$cc_hand_second_pair_or_pocket &&
f$cc_hand_third_pair_or_pocket

// Purpose: current no-made class. Draw equity no longer exists on River.
// Source/Provenance: T/A portable no-made evaluator.
##f$cc_river_delayed_no_made##
IsRiver && f$cc_hand_no_made

// Purpose: exact AK presence on the FLOP only, not later board cards.
// Source: native 3wBBvSB Gate13 skip exception checks Ace+King on flop.
// Source/Provenance: T direct persisted flop-card ranks.
##f$cc_river_delayed_flop_had_ak##
IsRiver &&
(FirstFlopCard = ace || SecondFlopCard = ace || ThirdFlopCard = ace) &&
(FirstFlopCard = king || SecondFlopCard = king || ThirdFlopCard = king)
