// Purpose: flop has ended and round-2 history is final.
// Source/Provenance: T betround numbering.
##f$cc_hist_flop_donk_closed##
IsTurn || IsRiver

// Purpose: Hero really executed the initial first-action Donk.
// Source: opportunity marker + no prior Hero check + closed aggressive counters.
// Provenance: A/T final-action proof.
##f$cc_hist_flop_donk_initial_bet_executed##
f$cc_hist_flop_donk_closed &&
user_cc_flop_donk_opportunity_seen &&
didchecround2 = 0 &&
(didbetsizeround2 > 0 || didalliround2 > 0)

// Purpose: initial Donk was a normal sized bet.
// Source/Provenance: T round-2 counter.
##f$cc_hist_flop_donk_initial_bet_was_sized##
f$cc_hist_flop_donk_initial_bet_executed && didbetsizeround2 > 0

// Purpose: initial Donk itself was direct all-in.
// Source/Provenance: T counters + A initial-action disambiguation.
##f$cc_hist_flop_donk_initial_bet_was_direct_allin##
f$cc_hist_flop_donk_initial_bet_executed &&
didalliround2 > 0 &&
didbetsizeround2 = 0 &&
didraisround2 = 0

// Purpose: Hero actually checked the initial Donk opportunity.
// Source/Provenance: T closed Hero check counter; later call/raise may coexist.
##f$cc_hist_flop_donk_initial_check_executed##
f$cc_hist_flop_donk_closed &&
user_cc_flop_donk_opportunity_seen &&
didchecround2 > 0

// Purpose: Hero Donked then faced a raise and called.
// Source/Provenance: T action counters + final aggressor identity.
##f$cc_hist_flop_donk_then_called_raise##
f$cc_hist_flop_donk_initial_bet_executed &&
didcallround2 > 0 &&
lastraised2 != userchair

// Purpose: Hero Donked then re-aggressed on flop.
// Source/Provenance: T multiple aggressive action counters.
##f$cc_hist_flop_donk_then_reaggressed##
f$cc_hist_flop_donk_initial_bet_executed &&
(didbetsizeround2 >= 2 || didraisround2 > 0 || (didbetsizeround2 > 0 && didalliround2 > 0))

// Purpose: initial Donk survived to next street with Hero still final flop aggressor.
// A Turn existing plus this one-bet/no-later-action pattern implies at least one
// opponent called; otherwise the hand would have ended on flop.
// Source/Provenance: T closed counters/lastraised2 + A call-survival inference.
##f$cc_hist_flop_donk_standard_called_parent##
f$cc_hist_flop_donk_closed &&
user_cc_flop_donk_opportunity_seen &&
didchecround2 = 0 &&
didcallround2 = 0 &&
didraisround2 = 0 &&
didbetsizeround2 = 1 &&
didalliround2 = 0 &&
lastraised2 = userchair

// Purpose: direct-all-in initial Donk reached a later street (rare runtime edge).
// Source/Provenance: T/A closed execution diagnostic.
##f$cc_hist_flop_donk_direct_allin_parent##
f$cc_hist_flop_donk_initial_bet_was_direct_allin && lastraised2 = userchair
