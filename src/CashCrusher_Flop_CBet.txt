// ============================================================================
// CashCrusher_Flop_CBet.txt
// Gate 01 — six-max cash flop CBet routing contract
// ============================================================================
// This file separates poker families BEFORE strategy is allowed to fire.
// It intentionally fails closed for child strategies not yet implemented.
//
// CashCrusher terminology:
//   CBet = Hero was final preflop aggressor in a raised pot and is checked to.
//   A bet in an unraised/limped pot is NOT routed here.
//
// Critical HU distinction:
//   TRUE HU DEAL (origin=1) != 3-6h deal reduced to HU by preflop folds (origin=2)
//   != a former multiway flop that later becomes HU (origin=3).
//
// Pot family separation:
//   ordinary SRP | HU limp-raised pot | ISO | ordinary 3BP | squeeze | 4BP | 5BP+
//
// Provenance:
//   Routing architecture = T/A from DeepCrusher + Gate00 context engine.
//   True-HU source ancestry = HUSB/HUBB families.
//   No final six-max betting frequency is encoded in this file yet.
// ============================================================================

// -----------------------------------------------------------------------------
// 01.01 — Gate ownership
// -----------------------------------------------------------------------------

##f$cc_flop_cbet_opportunity##
When !IsFlop Return false Force
When !f$cc_context_valid Return false Force
When AmountToCall > 0 Return false Force
When f$cc_pf_role_id <= 0 Return false Force
// Final preflop aggressor roles that may own a CBet.
When f$cc_pf_role_pfa Return true Force
When f$cc_pf_role_3bettor Return true Force
When f$cc_pf_role_4bettor Return true Force
When Others Return false Force

##f$cc_flop_cbet_router##
When !f$cc_flop_cbet_opportunity Return false Force

// -------------------------------------------------------------------------
// TRUE HU DEALS — never share a child policy with fold-reduced 3-6h HU.
// -------------------------------------------------------------------------

// One-raise ordinary HU SRP: SB is also BTN and PFA is IP postflop.
When f$cc_true_hu && f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_one_raise_ordinary_srp && f$cc_hu && f$cc_hu_ip
Return f$cc_flop_cbet_true_hu_srp_sb_pfa_ip Force

// SB limped, BB raised, SB called. This is a HU limp-raised pot, NOT ISO.
When f$cc_true_hu && f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_hu_limp_raise_proven && f$cc_hu && f$cc_hu_oop
Return f$cc_flop_cbet_true_hu_limpraised_bb_pfa_oop Force

// True-HU ordinary 3BP. Unknown/reversed limp-reraise chronology stays closed.

When f$cc_true_hu && f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_3bet_plain_proven && f$cc_hu && f$cc_hu_oop
Return f$cc_flop_cbet_true_hu_3bp_3bettor_oop Force

// True-HU 4BP remains mechanically distinct and strategically uncovered.
When f$cc_true_hu && f$cc_pot_family_id = 4 && f$cc_pf_role_4bettor && f$cc_hu && f$cc_hu_ip
Return f$cc_flop_cbet_true_hu_4bp_4bettor_ip Force
When f$cc_true_hu && f$cc_pot_family_id = 4 && f$cc_pf_role_4bettor && f$cc_hu && f$cc_hu_oop
Return f$cc_flop_cbet_true_hu_4bp_4bettor_oop Force

// -------------------------------------------------------------------------
// 3-6 HANDED DEALS — HU only if PRE-FLOP folds left two flop entrants.
// origin=3 (multiway flop later reduced to HU) is deliberately excluded.
// -------------------------------------------------------------------------

// One-raise pot: ordinary SRP versus proven ISO.
When f$cc_hu_origin_preflop_reduced && f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_iso_proven && f$cc_hu_ip
Return f$cc_flop_cbet_hu_iso_pfa_ip Force
When f$cc_hu_origin_preflop_reduced && f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_iso_proven && f$cc_hu_oop
Return f$cc_flop_cbet_hu_iso_pfa_oop Force

When f$cc_hu_origin_preflop_reduced && f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_one_raise_ordinary_srp && f$cc_hu_ip
Return f$cc_flop_cbet_hu_srp_pfa_ip Force
When f$cc_hu_origin_preflop_reduced && f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_one_raise_ordinary_srp && f$cc_hu_oop
Return f$cc_flop_cbet_hu_srp_pfa_oop Force

// Two-raise pot: proven squeeze versus proven plain 3BP. Unknown chronology
// never falls into plain 3BP by approximation.
When f$cc_hu_origin_preflop_reduced && f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_squeeze_proven && f$cc_hu_ip
Return f$cc_flop_cbet_hu_squeeze_3bettor_ip Force

Return f$cc_flop_cbet_hu_squeeze_3bettor_oop Force

Return f$cc_flop_cbet_hu_3bp_3bettor_ip Force
When f$cc_hu_origin_preflop_reduced && f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_3bet_plain_proven && f$cc_hu_oop
Return f$cc_flop_cbet_hu_3bp_3bettor_oop Force

// Three-raise pot: mechanical family exists, strategy remains isolated.
When f$cc_hu_origin_preflop_reduced && f$cc_pot_family_id = 4 && f$cc_pf_role_4bettor && f$cc_hu_ip
Return f$cc_flop_cbet_hu_4bp_4bettor_ip Force

Return f$cc_flop_cbet_hu_4bp_4bettor_oop Force

// -----------------------------------------------------------------------------
// 01.02 — Legacy ancestry metadata for HU ordinary SRP PFA
// -----------------------------------------------------------------------------
// Ancestor IDs:
//   1 HUSB       true HU Hero SB/Button, IP vs BB
//   2 HUBB       true HU Hero BB, OOP vs SB/Button
//   3 3wBTNvSB
//   4 3wBTNvBB
//   6 3wSBvBTN positional skeleton
//   7 3wSBvBB
//   0 no safe parent
//
// Numeric matchup IDs alone are insufficient. Matchup 56 means something very
// different in a TRUE HU deal (SB is Button/IP -> HUSB) than in a 3-6h deal
// reduced to SB-v-BB (SB is OOP -> 3wSBvBB).
// -----------------------------------------------------------------------------
