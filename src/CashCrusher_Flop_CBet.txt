// ============================================================================
// This file separates poker families BEFORE strategy is allowed to fire.
// It intentionally fails closed for child strategies not yet implemented.
//
// CashCrusher terminology:
//   CBet = Hero was final preflop aggressor in a raised pot and is checked to.
//   A bet in an unraised/limped pot is NOT routed here.
//
// Pot family separation:
//   ordinary SRP | ISO | ordinary 3BP | squeeze | 4BP | 5BP+
//
// Provenance:
//   Routing architecture = T/A from DeepCrusher + Gate00 context engine.
//   No final six-max betting frequency is encoded in this file yet.
// ============================================================================

// -----------------------------------------------------------------------------
// 01.01 — Gate ownership
// -----------------------------------------------------------------------------

##f$cc_flop_cbet_opportunity##
When !IsFlop Return false Force
When !f$cc_context_valid Return false Force
When AmountToCall > 0 Return false Force
When f$cc_pf_role_id <= 0 Return false Force
// Final preflop aggressor roles that may own a CBet.
When f$cc_pf_role_pfa Return true Force
When f$cc_pf_role_3bettor Return true Force
When f$cc_pf_role_4bettor Return true Force
When Others Return false Force

##f$cc_flop_cbet_router##
When !f$cc_flop_cbet_opportunity Return false Force

// One-raise pot: ordinary SRP versus proven ISO.
When f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_iso_proven && f$cc_hu && f$cc_hu_ip Return f$cc_flop_cbet_hu_iso_pfa_ip Force
When f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_iso_proven && f$cc_hu && f$cc_hu_oop Return f$cc_flop_cbet_hu_iso_pfa_oop Force
When f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && f$cc_pf_iso_proven && f$cc_multiway Return f$cc_flop_cbet_mw_iso_pfa Force

When f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && !f$cc_pf_iso_proven && f$cc_hu && f$cc_hu_ip Return f$cc_flop_cbet_hu_srp_pfa_ip Force
When f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && !f$cc_pf_iso_proven && f$cc_hu && f$cc_hu_oop Return f$cc_flop_cbet_hu_srp_pfa_oop Force
When f$cc_pot_family_id = 2 && f$cc_pf_role_pfa && !f$cc_pf_iso_proven && f$cc_multiway Return f$cc_flop_cbet_mw_srp_pfa Force

// Two-raise pot: proven squeeze versus proven plain 3BP. Unknown chronology
// never falls into the plain 3BP strategy by approximation.
When f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_squeeze_proven && f$cc_hu && f$cc_hu_ip Return f$cc_flop_cbet_hu_squeeze_3bettor_ip Force
When f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_squeeze_proven && f$cc_hu && f$cc_hu_oop Return f$cc_flop_cbet_hu_squeeze_3bettor_oop Force
When f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_squeeze_proven && f$cc_multiway Return f$cc_flop_cbet_mw_squeeze_3bettor Force

When f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_3bet_plain_proven && f$cc_hu && f$cc_hu_ip Return f$cc_flop_cbet_hu_3bp_3bettor_ip Force
When f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_3bet_plain_proven && f$cc_hu && f$cc_hu_oop Return f$cc_flop_cbet_hu_3bp_3bettor_oop Force
When f$cc_pot_family_id = 3 && f$cc_pf_role_3bettor && f$cc_pf_3bet_plain_proven && f$cc_multiway Return f$cc_flop_cbet_mw_3bp_3bettor Force

// Three-raise pot: mechanical family exists, strategy is deliberately isolated.
When f$cc_pot_family_id = 4 && f$cc_pf_role_4bettor && f$cc_hu && f$cc_hu_ip Return f$cc_flop_cbet_hu_4bp_4bettor_ip Force
When f$cc_pot_family_id = 4 && f$cc_pf_role_4bettor && f$cc_hu && f$cc_hu_oop Return f$cc_flop_cbet_hu_4bp_4bettor_oop Force
When f$cc_pot_family_id = 4 && f$cc_pf_role_4bettor && f$cc_multiway Return f$cc_flop_cbet_mw_4bp_4bettor Force

When Others Return false Force
