// -----------------------------------------------------------------------------
// C.01 — Hole-card high-card / blocker descriptors
// -----------------------------------------------------------------------------

// Purpose: expose whether Hero holds an ace for blocker/high-card selection.
// Source/Provenance: OpenPPL hole-card ranks, T mechanical descriptor.
##f$cc_hole_has_ace##
RankOfFirstHoleCard = ace || RankOfSecondHoleCard = ace

// Purpose: expose whether Hero holds a king for blocker/high-card selection.
// Source/Provenance: OpenPPL hole-card ranks, T mechanical descriptor.
##f$cc_hole_has_king##
RankOfFirstHoleCard = king || RankOfSecondHoleCard = king

// Purpose: expose whether Hero holds a queen; kept for future exact-combo work.
// Source/Provenance: OpenPPL hole-card ranks, T mechanical descriptor.
##f$cc_hole_has_queen##
RankOfFirstHoleCard = queen || RankOfSecondHoleCard = queen

// Purpose: expose whether Hero holds a jack; kept for future exact-combo work.
// Source/Provenance: OpenPPL hole-card ranks, T mechanical descriptor.
##f$cc_hole_has_jack##
RankOfFirstHoleCard = jack || RankOfSecondHoleCard = jack

// Purpose: count Hero hole cards in the T/J/Q/K/A broadway band.
// Source/Provenance: mechanical OpenPPL ranks, T/A convenience descriptor.
##f$cc_hole_broadway_count##
((RankOfFirstHoleCard >= ten) ? 1 : 0) + ((RankOfSecondHoleCard >= ten) ? 1 : 0)

// Purpose: identify two-broadway high-card holdings used in deterministic bluff
// selection on some static/high boards.
// Source/Provenance: A mechanical composition; betting relevance is P by caller.
##f$cc_hole_two_broadways##
f$cc_hole_broadway_count = 2

// -----------------------------------------------------------------------------
// C.02 — Made-hand CBet tiers
// -----------------------------------------------------------------------------

// Purpose: robust value bucket beginning at two pair.
// Source: DeepCrusher made-hand ladder.
// Provenance: A descriptor; whether to bet depends on context/board/SPR.
##f$cc_cbet_flop_very_strong_value##
f$cc_hand_two_pair_or_better

// Purpose: isolate an overpair that is not already two-pair+.
// Source: DeepCrusher overpair/value distinctions.
// Provenance: A descriptor only.
##f$cc_cbet_flop_overpair##
HaveOverPair && !f$cc_hand_two_pair_or_better

// Purpose: strong top-pair tier, using at most two unavailable better kicker ranks.
// Source: DeepCrusher kicker-quality concept.
// Provenance: A/P — source supports kicker stratification; exact <=2 boundary is
// a CashCrusher deterministic deep-stack split, not a universal solver cutoff.
##f$cc_cbet_flop_strong_top_pair##
f$cc_hand_just_top_pair && f$cc_number_better_kickers <= 2

// Purpose: middle top-pair kicker tier between strong and weak buckets.
// Source: DeepCrusher TP kicker hierarchy.
// Provenance: A/P — exact >2 and <=4 boundary is a CashCrusher mix abstraction.
##f$cc_cbet_flop_medium_top_pair##
f$cc_hand_just_top_pair && f$cc_number_better_kickers > 2 && f$cc_number_better_kickers <= 4

// Purpose: weakest top-pair kicker tier for pot-control/check-range decisions.
// Source: DeepCrusher TPWK concept.
// Provenance: A/P — >4 is an engineering split and remains matchup-adjustable.
##f$cc_cbet_flop_weak_top_pair##
f$cc_hand_just_top_pair && f$cc_number_better_kickers > 4

// Purpose: aggregate top pair, overpair and stronger made hands.
// Source/Provenance: A alias over portable DeepCrusher-derived primitives.
##f$cc_cbet_flop_top_pair_plus##
f$cc_hand_top_pair_or_better

// Purpose: aggregate lower pair / pocket-pair showdown value below top pair.
// Source: DeepCrusher MP/BP treatment.
// Provenance: A descriptor; no automatic CBet is implied.
##f$cc_cbet_flop_showdown_pair##
f$cc_hand_middle_or_bottom_pair && !f$cc_hand_top_pair_or_better

// -----------------------------------------------------------------------------
// C.03 — Draw tiers for deep-stack betting selection
// -----------------------------------------------------------------------------
// The old Spin label `StackOffDraws` is deliberately absent.  Draw quality is
// descriptive; later strategy chooses bet/check/XR/XC using position and SPR.

// Purpose: highest-equity/nut-potential no-made draw bucket.
// Source: DeepCrusher distinguishes combo, nut FD and strong draw+overcard cases.
// Provenance: A/P — ingredients are source-derived, exact grouping is a deep-stack
// CashCrusher replacement for the rejected StackOffDraws action shortcut.
##f$cc_cbet_flop_premium_draw##
f$cc_real_combo_draw ||
f$cc_real_nut_fd ||
(f$cc_real_oesd && f$cc_two_overcards) ||
(f$cc_real_fd && f$cc_two_overcards)

// Purpose: useful frontdoor draws below premium tier.
// Source: DeepCrusher good/medium draw distinctions.
// Provenance: A/P — deterministic grouping for six-max deep-stack selection.
##f$cc_cbet_flop_good_draw##
!f$cc_cbet_flop_premium_draw &&
(f$cc_real_fd || f$cc_real_oesd || (f$cc_real_gutshot && f$cc_two_overcards))

// Purpose: residual real frontdoor draws after premium/good tiers are removed.
// Source: DeepCrusher weak/medium draw architecture.
// Provenance: A/P; action is intentionally decided downstream.
##f$cc_cbet_flop_weak_draw##
!f$cc_cbet_flop_premium_draw && !f$cc_cbet_flop_good_draw && f$cc_real_frontdoor_draw

// -----------------------------------------------------------------------------
// C.04 — Air-combo quality
// -----------------------------------------------------------------------------
// Professional-theory fill (P): deterministic hardcoding needs principled bluff
// subsets.  Backdoors/high cards are selection features, not a claim that every
// such combo is solver-preferred in every matchup.

// Purpose: highest-quality air selector with both BDFD and BDSD.
// Source/Provenance: A backdoor ingredients + P deterministic bluff selection.
##f$cc_cbet_flop_air_best_backdoors##
f$cc_air_no_frontdoor && f$cc_combo_backdoors

// Purpose: air with at least one backdoor plus a meaningful high-card/blocker.
// Source/Provenance: P professional-theory selector built from T/A primitives.
// Limitation: exact blocker value remains board/matchup dependent.
##f$cc_cbet_flop_air_highcard_backdoor##
f$cc_air_no_frontdoor &&
(f$cc_backdoor_fd || f$cc_backdoor_sd) &&
(f$cc_hole_two_broadways || f$cc_hole_has_ace || f$cc_hole_has_king)

// Purpose: air with two overcards and at least one backdoor equity route.
// Source/Provenance: P deterministic mixed-strategy approximation.
##f$cc_cbet_flop_air_two_oc_backdoor##
f$cc_air_no_frontdoor && f$cc_two_overcards && (f$cc_backdoor_fd || f$cc_backdoor_sd)

// Purpose: union of reviewed high-quality air selectors available to CBet policy.
// Source/Provenance: P aggregation; context-specific child policy decides usage.
##f$cc_cbet_flop_quality_air##
f$cc_cbet_flop_air_best_backdoors || f$cc_cbet_flop_air_highcard_backdoor || f$cc_cbet_flop_air_two_oc_backdoor

// Purpose: residual air without the currently reviewed bluff-selection features.
// Source/Provenance: P negative descriptor; it does NOT mean automatic give-up on
// every future node/street.
##f$cc_cbet_flop_pure_air##
f$cc_air_no_frontdoor && !f$cc_cbet_flop_quality_air

// -----------------------------------------------------------------------------
// C.07 — High-SPR marginal-value caution
// -----------------------------------------------------------------------------
// DeepCrusher's <=16/20BB tactical thresholds are X.  Raw round-start SPR is

// Purpose: mark a materially deep postflop geometry for one-pair caution.
