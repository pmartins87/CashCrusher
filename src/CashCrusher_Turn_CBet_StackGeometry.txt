// CashCrusher_Turn_CBet_StackGeometry.txt
// Gate 02J.2A — mechanical Turn-CBet stack-sensitive execution geometry
// ============================================================================
// Purpose:
//   Measure the reviewed Turn bet against Hero balance and exact HU / dual-bound
//   multiway effective stacks without converting any historical Spin threshold
//   into strategy.
//
// Source findings retained for comparison:
//   * DeepCrusher frequently promoted a planned postflop size near ~60% Hero stack;
//   * its global callback could promote near ~50% effective stack;
//   * those thresholds arose in short-stack geometry and are NOT a universal
//     six-max cash Turn rule;
//   * in multiway, DeepCrusher f$EffectiveStack_BKP ancestry corresponds to the
//     deepest effective live relationship, not merely the shortest sidepot player.
//
// Source/Provenance:
//   T = OpenHoldem pot/currentbet/balance semantics and OpenPPL percent-pot sizing.
//   A = historical DeepCrusher 50/60 diagnostics and exact Turn size palette.
//   P = review classes only; no function here returns BetMax.
//
// Safety:
//   * this module is initial Turn-CBet geometry only (AmountToCall/currentbet=0);
//   * shortest-only multiway reach never means whole-field commitment;
//   * response to a later Turn raise remains defense-owned.
// ============================================================================

// Purpose: prove requested Turn CBet is a clean initial street bet whose simple
// fraction*street-start-pot geometry is valid.
// Source/Provenance: T OpenHoldem currentbet/potplayer/AmountToCall semantics.
##f$cc_turn_cbet_stackgeom_clean##
f$cc_turn_cbet_runtime_size_ready &&
f$cc_turn_cbet_runtime_mapping_valid &&
AmountToCall = 0 &&
currentbet = 0 &&
potplayer = 0 &&
f$cc_round_start_pot_bb > 0 &&
f$cc_hero_stack_round_start_bb > 0

// Purpose: requested Turn CBet contribution in big blinds.
// Source: T percent-pot initial-bet arithmetic; native betpot shares same clean
// no-call geometry for the reviewed fractions.
// Provenance: T/A runtime arithmetic over reviewed requested fraction.
##f$cc_turn_cbet_requested_bet_bb##
When !f$cc_turn_cbet_stackgeom_clean Return -1 Force
When f$cc_turn_cbet_requested_pot_fraction <= 0 Return -1 Force
When Others Return (f$cc_turn_cbet_requested_pot_fraction * f$cc_round_start_pot_bb) Force

// Purpose: Hero stack remaining after requested Turn bet, before any response.
// Source/Provenance: T stack arithmetic; negative means request exceeds balance.
##f$cc_turn_cbet_hero_residual_after_bet_bb##
When f$cc_turn_cbet_requested_bet_bb < 0 Return -1 Force
When Others Return (f$cc_hero_stack_round_start_bb - f$cc_turn_cbet_requested_bet_bb) Force

// Purpose: requested Turn bet / Hero available street-start stack.
// Source: T OpenHoldem balance-cap geometry.
// Provenance: T exact ratio in clean initial Turn-bet context.
##f$cc_turn_cbet_requested_hero_balance_ratio##
When !f$cc_turn_cbet_stackgeom_clean Return -1 Force
When f$cc_hero_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_cbet_requested_bet_bb / f$cc_hero_stack_round_start_bb) Force

// Purpose: historical DeepCrusher ~50% Hero/effective review signal only.
// Source/Provenance: A historical threshold diagnostic; NOT CashCrusher strategy.
##f$cc_turn_cbet_hist_hero_balance_50_trigger##
f$cc_turn_cbet_requested_hero_balance_ratio >= 0.50

// Purpose: historical explicit ~60%-of-Hero-stack sizing review signal only.
// Source/Provenance: A DeepCrusher diagnostic; does not authorize BetMax.
##f$cc_turn_cbet_hist_hero_stack_60_trigger##
f$cc_turn_cbet_requested_hero_balance_ratio >= 0.60

// Purpose: requested Turn bet / exact current HU effective stack.
// Source/Provenance: T/A exact HU street-start effective geometry.
##f$cc_turn_cbet_requested_hu_effective_ratio##
When !f$cc_turn_cbet_stackgeom_clean Return -1 Force
When !f$cc_hu Return -1 Force
When f$cc_hu_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_cbet_requested_bet_bb / f$cc_hu_effective_stack_round_start_bb) Force

// Purpose: historical ~50%-effective-stack HU review signal only.
// Source/Provenance: A DeepCrusher callback ancestry; no action follows here.
##f$cc_turn_cbet_hist_hu_effective_50_trigger##
f$cc_turn_cbet_requested_hu_effective_ratio >= 0.50

// Purpose: requested Turn bet / shallowest current multiway effective relation.
// Source/Provenance: T/A exact dual-bound multiway stack geometry.
// Safety: reaching this relation may create only a sidepot/all-in for one player.
##f$cc_turn_cbet_requested_mw_shallowest_effective_ratio##
When !f$cc_turn_cbet_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_shallowest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_cbet_requested_bet_bb / f$cc_mw_shallowest_effective_stack_round_start_bb) Force

// Purpose: requested Turn bet / deepest current multiway effective relation.
// Source/Provenance: T/A whole-field effective-depth denominator.
##f$cc_turn_cbet_requested_mw_deepest_effective_ratio##
When !f$cc_turn_cbet_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_deepest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_cbet_requested_bet_bb / f$cc_mw_deepest_effective_stack_round_start_bb) Force

// Purpose: historical DeepCrusher ~50%-effective multiway review signal using
// the deepest/all-live relationship rather than the shortest sidepot relation.
// Source/Provenance: A exact semantic ancestry; diagnostic only.
##f$cc_turn_cbet_hist_mw_effective_50_trigger##
f$cc_turn_cbet_requested_mw_deepest_effective_ratio >= 0.50

// Purpose: requested Turn size reaches Hero's full available stack.
// Source/Provenance: T mechanical balance geometry, not strategy.
##f$cc_turn_cbet_requested_reaches_hero_stack##
f$cc_turn_cbet_requested_hero_balance_ratio >= 1.00

// Purpose: requested Turn size reaches exact HU effective relationship.
// Source/Provenance: T exact HU effective-stack arithmetic.
##f$cc_turn_cbet_requested_reaches_hu_effective##
f$cc_hu && f$cc_turn_cbet_requested_hu_effective_ratio >= 1.00

// Purpose: requested Turn size reaches at least the shortest MW effective relation.
// Source/Provenance: T/A sidepot geometry descriptor.
##f$cc_turn_cbet_requested_reaches_mw_shallowest##
f$cc_multiway && f$cc_turn_cbet_requested_mw_shallowest_effective_ratio >= 1.00

// Purpose: requested Turn size reaches the deepest/all-live MW effective relation.
// Source/Provenance: T/A whole-field mechanical reach descriptor.
##f$cc_turn_cbet_requested_reaches_mw_deepest##
f$cc_multiway && f$cc_turn_cbet_requested_mw_deepest_effective_ratio >= 1.00

// Purpose: detect shortest-only sidepot reach while a deeper live relation remains.
// Source/Provenance: P/T diagnostic; this state must never trigger local BetMax.
##f$cc_turn_cbet_mw_sidepot_divergence_candidate##
f$cc_turn_cbet_requested_reaches_mw_shallowest && !f$cc_turn_cbet_requested_reaches_mw_deepest

// Purpose: compact review flag showing a Turn size engages historical or natural
// stack-sensitive geometry and therefore deserves audit/log visibility.
// Source/Provenance: A/P review metadata; true does NOT mean shove.
##f$cc_turn_cbet_stack_sensitive_review_required##
When !f$cc_turn_cbet_stackgeom_clean Return false Force
When f$cc_turn_cbet_hist_hero_balance_50_trigger Return true Force
When f$cc_hu && f$cc_turn_cbet_hist_hu_effective_50_trigger Return true Force
When f$cc_multiway && f$cc_turn_cbet_hist_mw_effective_50_trigger Return true Force
When f$cc_turn_cbet_requested_reaches_hero_stack Return true Force
When f$cc_hu && f$cc_turn_cbet_requested_reaches_hu_effective Return true Force
When f$cc_multiway && f$cc_turn_cbet_requested_reaches_mw_shallowest Return true Force
When Others Return false Force

// Purpose: bitfield-style diagnostic for source-threshold comparison.
// Bits: 1 Hero>=50; 2 Hero>=60; 4 HU-effective>=50; 8 reaches HU effective;
// 16 reaches MW shallowest; 32 MW-deepest>=50; 64 reaches MW deepest;
// 128 shortest-only sidepot divergence.
// Source/Provenance: P engineering metadata over A/T diagnostics.
##f$cc_turn_cbet_stack_review_flags_id##
When !f$cc_turn_cbet_stackgeom_clean Return 0 Force
When Others Return ((f$cc_turn_cbet_hist_hero_balance_50_trigger * 1) + (f$cc_turn_cbet_hist_hero_stack_60_trigger * 2) + ((f$cc_hu && f$cc_turn_cbet_hist_hu_effective_50_trigger) * 4) + ((f$cc_hu && f$cc_turn_cbet_requested_reaches_hu_effective) * 8) + ((f$cc_multiway && f$cc_turn_cbet_requested_reaches_mw_shallowest) * 16) + ((f$cc_multiway && f$cc_turn_cbet_hist_mw_effective_50_trigger) * 32) + ((f$cc_multiway && f$cc_turn_cbet_requested_reaches_mw_deepest) * 64) + ((f$cc_multiway && f$cc_turn_cbet_mw_sidepot_divergence_candidate) * 128)) Force

// ============================================================================
// End Gate 02J.2A geometry. No all-in action is encoded here.
