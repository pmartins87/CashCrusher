//   shallowest+deepest multiway effective relationships. This module describes

// Source/Provenance:
//   T = OpenHoldem/OpenPPL pot/currentbet/balance geometry.
//   A = same dual-bound stack correction frozen for CBet streets.
//   P = sidepot-safe Float diagnostics.

// Purpose: prove simple fraction*pot Float geometry is valid before execution.
// Source: canonical Float is Hero's first flop action with no bet facing Hero.
// Provenance: T exact currentbet/potplayer/AmountToCall runtime guard.
##f$cc_flop_float_stackgeom_clean##
f$cc_flop_float_runtime_size_ready &&
f$cc_flop_float_runtime_mapping_valid &&
AmountToCall = 0 &&
currentbet = 0 &&
potplayer = 0 &&
f$cc_round_start_pot_bb > 0 &&
f$cc_hero_stack_round_start_bb > 0

// Purpose: requested Float contribution in big blinds.
// Source/Provenance: T exact initial-bet percentage arithmetic.
##f$cc_flop_float_requested_bet_bb##
When !f$cc_flop_float_stackgeom_clean Return -1 Force
When f$cc_flop_float_requested_pot_fraction <= 0 Return -1 Force
When Others Return (f$cc_flop_float_requested_pot_fraction * f$cc_round_start_pot_bb) Force

// Purpose: requested Float / Hero available flop-start stack.
// Source/Provenance: T mechanical balance ratio.
##f$cc_flop_float_requested_hero_balance_ratio##
When !f$cc_flop_float_stackgeom_clean Return -1 Force
When f$cc_hero_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_flop_float_requested_bet_bb / f$cc_hero_stack_round_start_bb) Force

// Purpose: requested Float / exact current HU effective stack.
// Source/Provenance: T/A HU effective geometry.
##f$cc_flop_float_requested_hu_effective_ratio##
When !f$cc_flop_float_stackgeom_clean Return -1 Force
When !f$cc_hu Return -1 Force
When f$cc_hu_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_flop_float_requested_bet_bb / f$cc_hu_effective_stack_round_start_bb) Force

// Purpose: requested Float / shallowest multiway effective relation.
// Source/Provenance: T/A dual-bound multiway geometry.
// Safety: reaching this denominator may mean only one sidepot player is all-in.
##f$cc_flop_float_requested_mw_shallowest_ratio##
When !f$cc_flop_float_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_shallowest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_flop_float_requested_bet_bb / f$cc_mw_shallowest_effective_stack_round_start_bb) Force

// Purpose: requested Float / deepest all-live multiway effective relation.
// Source/Provenance: T/A whole-field effective-depth denominator.
##f$cc_flop_float_requested_mw_deepest_ratio##
When !f$cc_flop_float_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_deepest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_flop_float_requested_bet_bb / f$cc_mw_deepest_effective_stack_round_start_bb) Force

// Purpose: reviewed Float size already exhausts Hero stack.
// Source/Provenance: T mechanical balance reach.
##f$cc_flop_float_requested_reaches_hero_stack##
f$cc_flop_float_requested_hero_balance_ratio >= 1.00

// Purpose: reviewed Float size reaches exact HU effective relation.
// Source/Provenance: T/A HU mechanical reach.
##f$cc_flop_float_requested_reaches_hu_effective##
f$cc_hu && f$cc_flop_float_requested_hu_effective_ratio >= 1.00

// Purpose: reviewed Float reaches at least one shallow multiway relationship.
// Source/Provenance: T/A sidepot descriptor only.
##f$cc_flop_float_requested_reaches_mw_shallowest##
f$cc_multiway && f$cc_flop_float_requested_mw_shallowest_ratio >= 1.00

// Purpose: reviewed Float reaches the deepest/all-live multiway relation.
// Source/Provenance: T/A whole-field mechanical reach.
##f$cc_flop_float_requested_reaches_mw_deepest##
f$cc_multiway && f$cc_flop_float_requested_mw_deepest_ratio >= 1.00

// Purpose: detect a short sidepot all-in while deeper opponents remain matchable.
// Source/Provenance: P/T sidepot safety diagnostic.
##f$cc_flop_float_mw_sidepot_divergence_candidate##
f$cc_flop_float_requested_reaches_mw_shallowest && !f$cc_flop_float_requested_reaches_mw_deepest

// Purpose: historical near-all-in review signal only, not an action.
// Source: DeepCrusher repeatedly used ~50/55/60% balance/effective thresholds in
// short-stack sizing callbacks and XR plans.
// Provenance: A diagnostic retained for exact-node review, never generic shove.
##f$cc_flop_float_historical_near_allin_review##
When !f$cc_flop_float_stackgeom_clean Return false Force
When f$cc_flop_float_requested_hero_balance_ratio >= 0.50 Return true Force
When f$cc_hu && f$cc_flop_float_requested_hu_effective_ratio >= 0.50 Return true Force
When f$cc_multiway && f$cc_flop_float_requested_mw_deepest_ratio >= 0.50 Return true Force
When Others Return false Force
