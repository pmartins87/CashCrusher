// Purpose: exact Turn top-pair-or-better current class.
// Source/Provenance: T/A portable hand hierarchy.
##f$cc_turn_probe_tpplus##
IsTurn && f$cc_hand_top_pair_or_better

// Purpose: exact current second-pair/pocket tier below TP.
// Source/Provenance: T/A portable pair ladder.
##f$cc_turn_probe_second_pair##
IsTurn && f$cc_hand_second_pair_or_pocket && !f$cc_hand_top_pair_or_better

// Purpose: source low-pair union: current 3rd/4th/5th pair/pocket tiers.
// Source: newest HUBB Turn-Probe tree uses these exact families with `TurnMin`.
// Provenance: T/A portable pair ladder.
##f$cc_turn_probe_low_pair##
IsTurn &&
!f$cc_hand_top_pair_or_better &&
!f$cc_hand_second_pair_or_pocket &&
(f$cc_hand_third_pair_or_pocket || f$cc_hand_fourth_pair_or_pocket || f$cc_hand_fifth_pair_or_pocket)

// Purpose: current no-made hand with real frontdoor Gutshot/OESD/FD equity.
// Source: DeepCrusher `NoMadeHand && Gutshot_or_BetterFDSD`.
// Provenance: A exact concept over cleaned portable real-draw evaluators.
##f$cc_turn_probe_live_frontdoor_draw##
IsTurn && f$cc_hand_no_made && f$cc_real_frontdoor_draw

// Purpose: source-compatible air: no made hand and no Gutshot-or-better real draw.
// Source/Provenance: T/A equivalent of generic DeepCrusher `f$hand_Air` concept.
##f$cc_turn_probe_air##
IsTurn && f$cc_air_no_frontdoor
