// Purpose: source 2-hole-card straight contribution.
// Source: DeepCrusher `f$hand_2HC_Straight` exact definition.
// Provenance: T mechanical descriptor.
##f$cc_river_donk_2hc_straight##
HaveStraight && (nstraightfillcommon - nstraightfill = 2)

// Purpose: source 2-hole-card flush contribution.
// Source: DeepCrusher `f$hand_2HC_Flush` exact definition.
// Provenance: T mechanical descriptor.
##f$cc_river_donk_2hc_flush##
HaveFlush && SuitsInHand = 1 && nsuitedcommon = 3

// Purpose: source one-hole-card straight class.
// Source/Provenance: T exact complement of 2HC straight.
##f$cc_river_donk_1hc_straight##
HaveStraight && !f$cc_river_donk_2hc_straight

// Purpose: source one-hole-card flush class.
// Source/Provenance: T exact complement of 2HC flush.
##f$cc_river_donk_1hc_flush##
HaveFlush && !f$cc_river_donk_2hc_flush

// Purpose: one-card straight that is the under-straight class.
// Source: `f$HaveOneCardUnderStraight = f$hand_1HC_Straight && HaveUnderStraight`.
// Provenance: T.
##f$cc_river_donk_1hc_understraight##
f$cc_river_donk_1hc_straight && HaveUnderStraight

// Purpose: one-card straight that is not under-straight.
// Source: exact DeepCrusher helper.
// Provenance: T.
##f$cc_river_donk_1hc_straight_not_under##
f$cc_river_donk_1hc_straight && !HaveUnderStraight

// Purpose: K-high-or-better contributing card in a 1HC flush.
// Source: DeepCrusher `f$KingHighContributingToOnlyOneCardFlush`.
// Provenance: T; rewritten as flat WHENs so indentation has no semantics.
##f$cc_river_donk_1hc_flush_khigh_or_better##
When !f$cc_river_donk_1hc_flush Return false Force
When nsuitedcommon > 3 && tsuitcommon = 0 && (hand$Ah || hand$Kh) Return true Force
When nsuitedcommon > 3 && tsuitcommon = 1 && (hand$Ad || hand$Kd) Return true Force
When nsuitedcommon > 3 && tsuitcommon = 2 && (hand$Ac || hand$Kc) Return true Force
When nsuitedcommon > 3 && tsuitcommon = 3 && (hand$As || hand$Ks) Return true Force
When nsuitedcommon > 3 && NumberOfUnknownSuitedOvercards <= 1 Return true Force
When Others Return false Force

// Purpose: Q/J/T/9 contributing card in the one-card flush suit.
// Source: DeepCrusher `f$NineToQueenContributingToOnlyOneCardFlush`.
// Provenance: T, flat parser-safe translation.
##f$cc_river_donk_1hc_flush_9_to_q##
When !f$cc_river_donk_1hc_flush Return false Force
When SuitsInHand = 2 && tsuitcommon = 0 && (hand$Qh || hand$Jh || hand$Th || hand$9h) Return true Force
When SuitsInHand = 2 && tsuitcommon = 1 && (hand$Qd || hand$Jd || hand$Td || hand$9d) Return true Force
When SuitsInHand = 2 && tsuitcommon = 2 && (hand$Qc || hand$Jc || hand$Tc || hand$9c) Return true Force
When SuitsInHand = 2 && tsuitcommon = 3 && (hand$Qs || hand$Js || hand$Ts || hand$9s) Return true Force
When Others Return false Force

// Purpose: <=8 contributing 1HC flush bucket after K+ and 9-Q buckets.
// Source: DeepCrusher LowMiddle helper plus source branch precedence.
// Provenance: T/A conservative disjoint bucket.
##f$cc_river_donk_1hc_flush_low##
f$cc_river_donk_1hc_flush &&
!f$cc_river_donk_1hc_flush_khigh_or_better &&
!f$cc_river_donk_1hc_flush_9_to_q

// Purpose: source river-completed descriptor.
// Source: DeepCrusher `f$river_Completed = FlushPossible || StraightPossible`.
// Provenance: T.
##f$cc_river_donk_completed##
FlushPossible || StraightPossible

// Purpose: source river-super-completed descriptor retained for later branches.
// Source: DeepCrusher exact definition.
// Provenance: T.
##f$cc_river_donk_super_completed##
FlushPossible && StraightPossible

// Purpose: source River underboard descriptor needed once Turn-call provenance is
// written by the future defense gate.
// Source: DeepCrusher `f$river_UnderBoard` exact definition.
// Provenance: T mechanical only.
##f$cc_river_donk_underboard##
(RiverCard < LowestFlopCard) && (RiverCard < TurnCard)

// Purpose: conservative exact-two-pair contribution subset that excludes a board
// already showing two pair; avoids treating board-only two pair as Hero value.
// Source: DeepCrusher uses contribution-aware CF7_ExactTwoPair.
// Provenance: A conservative subset pending a direct CF7 equivalent.
##f$cc_river_donk_contributed_exact_two_pair##
HaveTwoPair && !TwoPairOnBoard

// Purpose: literal contributed set is directly reconstructible without CF7.
// Source/Provenance: T OpenPPL HaveSet; subset of source `(CF7_RealTrips||HaveSet)`.
##f$cc_river_donk_source_set##

// Source precedence mirrors mature DeepCrusher Retro09.

// non-completed needs prior-Turn provenance for the 100 exception, otherwise check.

// Prior TP/OP, prior 2P+, missed-good-draw bluff and MP/BP+draw branches are NOT
// reconstructed from current River strength. They remain blocked until defense.
