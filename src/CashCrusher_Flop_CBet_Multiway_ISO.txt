// Purpose: choose ISO CBet/check when Hero acts LAST after two checks.
// Source/Provenance: A multiway/ISO concepts + P exact three-way cash policy.
// Decision logic: widest pressure versus two limpers; mixed composition tighter;
// two-coldcaller family most selective; robust value/premium draws remain active;
// air requires best backdoors on favorable static high board and weak composition.
##f$cc_cbet_mw_iso_true3_last_action##
When !f$cc_mw_iso_true3_last_context Return false Force
When !f$cc_mw_iso_true3_context_consistent Return false Force
When f$cc_cbet_flop_very_strong_value Return true Force
When f$cc_cbet_flop_overpair && f$cc_flop_dynamic_lowmid_parent && f$cc_mw_iso_true3_two_coldcallers && f$cc_mw_cbet_high_spr Return false Force
When f$cc_cbet_flop_overpair Return true Force
When f$cc_cbet_flop_strong_top_pair && f$cc_flop_dynamic_lowmid_parent && f$cc_mw_iso_true3_two_coldcallers && f$cc_mw_cbet_high_spr Return false Force
When f$cc_cbet_flop_strong_top_pair Return true Force
When f$cc_cbet_flop_medium_top_pair && f$cc_pair_plus_draw Return true Force
When f$cc_cbet_flop_medium_top_pair && f$cc_flop_static_high_parent && f$cc_mw_iso_true3_two_limpers Return true Force
When f$cc_cbet_flop_medium_top_pair && f$cc_flop_static_high_parent && f$cc_mw_iso_true3_limper_plus_coldcaller && f$cc_mw_all_effective_spr_below_4 Return true Force
When f$cc_cbet_flop_medium_top_pair Return false Force
When f$cc_cbet_flop_weak_top_pair Return false Force
When f$cc_hand_second_pair_or_pocket && f$cc_pair_plus_draw && f$cc_mw_iso_true3_two_limpers Return true Force
When f$cc_cbet_flop_showdown_pair Return false Force
When f$cc_cbet_flop_premium_draw Return true Force
When f$cc_cbet_flop_good_draw && (f$cc_flop_static_high_parent || f$cc_flop_paired_parent) && !f$cc_mw_iso_true3_two_coldcallers Return true Force
When f$cc_cbet_flop_good_draw && f$cc_flop_dynamic_high_parent && f$cc_mw_iso_true3_two_limpers Return true Force
When f$cc_cbet_flop_good_draw && f$cc_flop_dynamic_lowmid_parent && f$cc_mw_iso_true3_two_limpers && f$cc_two_overcards Return true Force
When f$cc_real_frontdoor_draw Return false Force
When f$cc_cbet_flop_air_best_backdoors && f$cc_flop_static_high_parent && f$cc_mw_iso_true3_two_limpers Return true Force
When f$cc_cbet_flop_air_best_backdoors && f$cc_flop_paired_parent && f$cc_mw_iso_true3_two_limpers && f$cc_mw_all_effective_spr_below_4 Return true Force
When Others Return false Force
