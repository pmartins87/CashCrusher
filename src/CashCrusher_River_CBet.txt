// Purpose: ordinary SRP Turn families not matched by a direct source descendant.
// Source/Provenance: P exact residual-family metadata; Gate03C owns reviewed gaps.
##f$cc_river_srp_residual_context##
f$cc_river_cbet_opportunity &&
(f$cc_river_turn_family_id = 1 ||
 f$cc_river_turn_family_id = 2 ||
 f$cc_river_turn_family_id = 3 ||
 f$cc_river_turn_family_id = 4 ||
 f$cc_river_turn_family_id = 7 ||
 f$cc_river_turn_family_id = 8) &&
!f$cc_river_srp_source_covered

// Purpose: current River descends from ISO Turn-CBet family.
// Source/Provenance: P exact Gate02 family carry-forward; Gate03D owns policy.
##f$cc_river_iso_context##
f$cc_river_cbet_opportunity && f$cc_river_turn_family_id = 9

// Purpose: current River descends from plain-3BP Turn-CBet family.
// Source/Provenance: P exact Gate02 family carry-forward; Gate03E owns policy.
##f$cc_river_plain3bp_context##
f$cc_river_cbet_opportunity && f$cc_river_turn_family_id = 10

// Purpose: current River descends from squeeze Turn-CBet family.
// Source/Provenance: P exact Gate02 family carry-forward; Gate03F owns policy.
##f$cc_river_squeeze_context##

// Purpose: current River descends from clean supported HU 4BP Turn-CBet family.
// Source/Provenance: P exact Gate02 family carry-forward; Gate03G owns policy.
##f$cc_river_4bp_context##
f$cc_river_cbet_opportunity && f$cc_river_turn_family_id = 12

// Purpose: compact River-CBet family ID for logs/coverage.
// IDs: 1 source-SRP; 2 P-heavy SRP; 3 ISO; 4 plain3BP; 5 squeeze; 6 4BP;
// 0 unknown/invalid.
// Source/Provenance: P engineering metadata; never strategy alone.
##f$cc_river_cbet_family_id##
When !f$cc_river_cbet_opportunity Return 0 Force
When f$cc_river_srp_source_covered Return 1 Force
When f$cc_river_srp_gap_covered Return 2 Force
When f$cc_river_iso_covered Return 3 Force
When f$cc_river_plain3bp_covered Return 4 Force
When f$cc_river_squeeze_covered Return 5 Force
When f$cc_river_4bp_covered Return 6 Force
When f$cc_river_srp_residual_context Return 2 Force
When f$cc_river_iso_context Return 3 Force
When f$cc_river_plain3bp_context Return 4 Force
When f$cc_river_squeeze_context Return 5 Force
When f$cc_river_4bp_context Return 6 Force
When Others Return 0 Force

// Purpose: top-level River CBet action router.
// Source: DeepCrusher normal-CBet quarantine: unmatched genuine 2Bar histories
// check instead of falling through to broad tails.
// Provenance: A/P exact-family fail-closed router.
##f$cc_river_cbet_router##
When !f$cc_river_cbet_opportunity Return false Force
When f$cc_river_srp_source_covered Return f$cc_river_srp_source_action Force
When f$cc_river_srp_gap_covered Return f$cc_river_srp_gap_action Force
When f$cc_river_iso_covered Return f$cc_river_iso_action Force
When f$cc_river_plain3bp_covered Return f$cc_river_plain3bp_action Force
When f$cc_river_squeeze_covered Return f$cc_river_squeeze_action Force
When f$cc_river_4bp_covered Return f$cc_river_4bp_action Force
When Others Return false Force

// Purpose: paired River strategic size-ID router.
// Source/Provenance: A/P exact owner; unsupported/check ->0.
##f$cc_river_cbet_size_id##
When !f$cc_river_cbet_opportunity Return 0 Force
When f$cc_river_srp_source_covered Return f$cc_river_srp_source_size_id Force
When f$cc_river_srp_gap_covered Return f$cc_river_srp_gap_size_id Force
When f$cc_river_iso_covered Return f$cc_river_iso_size_id Force
When f$cc_river_plain3bp_covered Return f$cc_river_plain3bp_size_id Force
When f$cc_river_squeeze_covered Return f$cc_river_squeeze_size_id Force
When f$cc_river_4bp_covered Return f$cc_river_4bp_size_id Force
When Others Return 0 Force

// Purpose: report whether current canonical River-CBet family has implemented
// strategic coverage rather than merely a recognized topology.
// Source/Provenance: P engineering coverage contract.
##f$cc_river_cbet_strategy_covered##
f$cc_river_srp_source_covered ||
f$cc_river_srp_gap_covered ||
f$cc_river_iso_covered ||
f$cc_river_plain3bp_covered ||
f$cc_river_squeeze_covered ||
f$cc_river_4bp_covered

// Purpose: positive action must own positive size; check/uncovered owns size0.
// Source/Provenance: P safety invariant.
##f$cc_river_cbet_size_consistent##
When !f$cc_river_cbet_router Return f$cc_river_cbet_size_id = 0 Force
When f$cc_river_cbet_router Return f$cc_river_cbet_size_id > 0 Force
When Others Return false Force
