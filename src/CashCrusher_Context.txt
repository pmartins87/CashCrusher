// ============================================================================
// CashCrusher_Context.txt
// Gate 00C/00F — dynamic 2-to-6-handed post-flop context engine
// ============================================================================
// Scope:
//   * support a six-max cash table when 2, 3, 4, 5 or 6 players are dealt in;
//   * preserve a stable six-max-equivalent positional taxonomy;
//   * distinguish TRUE HU DEALS from hands that merely become HU after folds;
//   * preserve how many players actually reached the flop;
//   * preserve exact absolute positions and live-opponent mask;
//   * classify HU / multiway and relative post-flop position;
//   * classify preflop pot family from OpenPPL historical raise count;
//   * classify Hero's preflop role from stable historical actions;
//   * prove ISO / HU limp-raise / squeeze subtypes only when history supports it;
//   * fail closed on contradictory mechanical states.
//
// IMPORTANT STRATEGIC DISTINCTION:
//   f$cc_hu only means TWO PLAYERS ARE CURRENTLY PLAYING.
//   It does NOT say how the hand began.
//
//   f$cc_hu_origin_id = 1  TRUE_HU_DEAL
//       Exactly two players were dealt into the hand. HU button/SB rules apply.
//       This is the natural descendant of legacy HUSB/HUBB knowledge.
//
//   f$cc_hu_origin_id = 2  PREFLOP_REDUCED_TO_HU
//       3-6 players were dealt, but only two reached the flop after preflop folds.
//       Absolute 6-max-equivalent positions and ranges remain relevant.
//
//   f$cc_hu_origin_id = 3  POSTFLOP_REDUCED_TO_HU
//       3+ players reached the flop and later action reduced the hand to HU.
//       Future streets MUST retain multiway provenance; ordinary HU policy must
//       not silently take over.
//
// Stable position IDs:
//   1 UTG | 2 HJ | 3 CO | 4 BTN | 5 SB | 6 BB
//
// Canonical position mapping by dealt-player count:
//   6h: UTG HJ CO BTN SB BB
//   5h:     HJ CO BTN SB BB
//   4h:        CO BTN SB BB
//   3h:           BTN SB BB
//   2h:               SB BB   (SB is also BTN/dealer by HU rules)
//
// Live-position mask bits:
//   UTG=1 | HJ=2 | CO=4 | BTN=8 | SB=16 | BB=32
// In a true HU deal the dealer/SB is represented ONLY by SB bit 16, never
// double-counted as BTN bit 8.
//
// Pot-family IDs:
//   1 UNRAISED | 2 ONE_RAISE | 3 THREE_BET | 4 FOUR_BET | 5 FIVE_BET_PLUS
// Hero-role IDs:
//   1 FINAL_AGGRESSOR_1R (PFA)
//   2 CALLER_1R
//   3 FINAL_AGGRESSOR_2R (3bettor)
//   4 OPEN_RAISER_CALL_3BET
//   5 COLD_CALL_3BET
//   6 FINAL_AGGRESSOR_3R (4bettor)
//   7 CALL_4BET_GENERIC
//   8 UNRAISED_CALLER/LIMPER
//   9 UNRAISED_BB_CHECK
// Relative-position IDs:
//   1 FIRST | 2 MIDDLE | 3 LAST
//
// Provenance contract:
//   T = direct OpenHoldem/OpenPPL mechanical fact.
//   A = structural adaptation of audited DeepCrusher scenario/history concepts.
//   P = six-max engineering inference where the source has no literal state.
// This file classifies context only; it contains no strategic bet frequency.
// ============================================================================

// 00F.01 — Dynamic table/deal handedness

// Purpose: number of players actually dealt into this hand, limited to supported 2-6h.
// Source/Provenance: OpenPPL `nplayersdealt`; T mechanical fact.
##f$cc_deal_size##
When nplayersdealt >= 2 && nplayersdealt <= 6 Return nplayersdealt Force
When Others Return 0 Force

// Purpose: boolean guard for the supported CashCrusher handedness domain.
// Source/Provenance: A alias over `f$cc_deal_size`.
##f$cc_supported_deal##
f$cc_deal_size >= 2 && f$cc_deal_size <= 6

// Purpose: count physically seated players, which may exceed players actually dealt.
// Source/Provenance: OpenPPL `OpponentsAtTable`/`nopponentsseated`; T.
##f$cc_table_seated_count##
When OpponentsAtTable < 1 Return 0 Force
When Others Return (OpponentsAtTable + 1) Force

// Purpose: reject impossible seated/dealt count combinations before range routing.
// Source/Provenance: P engineering consistency invariant over T player counts.
##f$cc_table_count_consistent##
When f$cc_table_seated_count < 2 Return false Force
When f$cc_table_seated_count > 6 Return false Force
When f$cc_deal_size <= 0 Return false Force
When f$cc_table_seated_count < f$cc_deal_size Return false Force
When Others Return true Force

// Purpose: identify a physical table with only two seats currently occupied.
// Source/Provenance: T/A alias over seated-player count; not a hand-range origin by itself.
##f$cc_physical_table_hu##
f$cc_table_seated_count = 2

// Purpose: identify a hand in which exactly two players were actually dealt cards.
// Source/Provenance: T/A alias over dealt-player count.
##f$cc_deal_hu##
f$cc_deal_size = 2

// Purpose: strategic true-HU origin flag: only two players were dealt into the hand.
// Source/Provenance: A replacement for blindly using legacy HUSB/HUBB table labels.
##f$cc_true_hu##
f$cc_deal_hu

// Purpose: distinguish a true-HU hand on a physically two-player table.
// Source/Provenance: P/A diagnostic composition of true-HU and seated count.
##f$cc_true_hu_full_table##
f$cc_true_hu && f$cc_physical_table_hu

// Purpose: true-HU hand even though extra seats remain shown/seated but were not dealt.
// Source/Provenance: P/A handedness distinction required for dynamic cash tables.
##f$cc_true_hu_with_waiters##
f$cc_true_hu && f$cc_table_seated_count > 2

// Purpose: explicit six-handed deal descriptor for audit/range modifiers.

// -----------------------------------------------------------------------------
// 00F.04 — Who actually reached the flop?
// -----------------------------------------------------------------------------
// playersdealtbits persists the hand-start deal. foldbits1 is the persisted
// preflop fold mask. Clearing those bits gives the exact preflop survivors.
// This is critical on turn/river: a 3-way flop that later becomes HU must not
// be rewritten into an ordinary HU hand.

// Purpose: ensure persisted preflop folds are a subset of players actually dealt.
// Source/Provenance: OpenPPL `foldbits1`/`playersdealtbits`; T/P bitmask sanity check.
##f$cc_preflop_fold_bits_valid##
When ((foldbits1 BitAnd playersdealtbits) != foldbits1) Return false Force
When Others Return true Force

// Purpose: reconstruct exact set of players that survived preflop and reached flop.
// Source/Provenance: T persisted dealt/fold bitmasks; A context reconstruction.
##f$cc_flop_entry_bits##
playersdealtbits - (playersdealtbits BitAnd foldbits1)

// Purpose: exact number of players who entered the flop, preserved on later streets.
// Source/Provenance: T `BitCount` over reconstructed flop-entry mask.
##f$cc_flop_entry_count##
When IsPreflop Return 0 Force
When Others Return BitCount(f$cc_flop_entry_bits) Force

// Purpose: reject impossible flop-entry counts and later live counts.
// Source/Provenance: P mechanical consistency invariant over T persisted masks.
##f$cc_flop_entry_valid##
When IsPreflop Return false Force
When f$cc_flop_entry_count < 2 Return false Force
When f$cc_flop_entry_count > f$cc_deal_size Return false Force
When nplayersplaying > f$cc_flop_entry_count Return false Force
When Others Return true Force

// Purpose: exactly two players reached the flop, regardless of original deal size.
// Source/Provenance: A alias over exact flop-entry count.
##f$cc_reached_flop_hu##
f$cc_flop_entry_count = 2

// Purpose: three or more players reached the flop; provenance must survive later folds.
// Source/Provenance: A multiway-origin alias over exact flop-entry count.
##f$cc_reached_flop_multiway##

// Purpose: current HU came from a 3-6h deal reduced to two before flop.
// Source/Provenance: P/A key range-topology distinction absent from Spin scenario labels.
##f$cc_preflop_reduced_hu##
f$cc_hu && f$cc_deal_size >= 3 && f$cc_reached_flop_hu

// Purpose: current HU arose only after a multiway flop lost players postflop.
// Source/Provenance: P/A provenance guard preventing ordinary HU-policy leakage.
##f$cc_postflop_reduced_to_hu##
f$cc_hu && f$cc_reached_flop_multiway
