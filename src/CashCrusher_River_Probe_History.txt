// Gate 11A / 11A.1 — canonical River Probe ownership from CLOSED flop X/C -> turn X/X
// ============================================================================
// Purpose:
//   Prove that Hero reaches a River PROBE after checking and calling exactly once
//   on flop, then checking a Turn that checks through, and finally reaching its
//   first River action FIRST/MIDDLE with no wager facing.
//
// Gate11A.1 source-fidelity correction:
//   Crusher Framework routes FIRST/MIDDLE River no-action states by the ACTUAL
//   flop history: `BotCalledOnFlop -> f$move_river_Probe`; `!BotCalledOnFlop ->
//   f$river_lead_noaction`. It does NOT require Hero to lack preflop initiative.
//   Therefore an opener/PFA who checked and then ACTUALLY X/C'd flop is a River-
//   Probe history too. Preflop role still matters to each strategic child/range,
//   but it must not delete the generic River-Probe owner itself.
//
// Ownership boundary:
//   * flop X/C -> turn X/C -> River lead is River Donk (Gate09), not Probe;
//   * flop X/X -> Turn lead is Turn Probe (Gate10), not River Probe;
//   * flop X/C -> turn X/X is River Probe regardless of preflop initiative;
//   * exact LAST is a delayed/stab owner, not this FIRST/MIDDLE Probe family.
//
// Source boundary:
//   Crusher Framework 5 leaves `f$move_river_Probe` empty. Positive strategy must
//   therefore come from explicit Starting-Strategy/TBP/DeepCrusher descendants or
//   separately labeled professional six-max fills. Generic ownership grants no bet.
//
// Source/Provenance:
//   T = OpenHoldem closed round-2/3 counters, raisbits/lastraised and River facts.
//   A = audited Framework/DeepCrusher Probe-vs-Donk/noaction ownership translation.
//   P = strict field/position/history firewalls and 5bet+ fail-closed boundary.
// ============================================================================

// -----------------------------------------------------------------------------
// RP-H.01 — exact flop X/C proof
// -----------------------------------------------------------------------------

// Purpose: exactly one opponent produced flop aggression.
// Source/Provenance: T OpenHoldem persisted raisbits2.
##f$cc_river_probe_flop_unique_aggressor##
BitCount(raisbits2) = 1

// Purpose: final flop aggressor is a valid non-Hero chair and agrees with mask.
// Source/Provenance: T independent lastraised2 + raisbits2 cross-check.
##f$cc_river_probe_flop_aggressor_valid##
When !f$cc_river_probe_flop_unique_aggressor Return false Force
When lastraised2 < 0 || lastraised2 > 9 Return false Force
When lastraised2 = userchair Return false Force
When (raisbits2 BitAnd (1 << lastraised2)) <= 0 Return false Force
When Others Return true Force

// Purpose: Hero executed clean flop X/C and never bet/raised/jammed.
// Source: mature DeepCrusher Retro11 markers are written only after final CALL;
// this closed-history proof is the CashCrusher mechanical equivalent.
// Source/Provenance: T/A.
##f$cc_hist_river_probe_flop_checkcall_clean##
When !IsRiver Return false Force
When didchecround2 <= 0 Return false Force
When didcallround2 != 1 Return false Force
When didraisround2 != 0 Return false Force
When didbetsizeround2 != 0 Return false Force
When didalliround2 != 0 Return false Force
When !f$cc_river_probe_flop_aggressor_valid Return false Force
When Others Return true Force

// -----------------------------------------------------------------------------
// RP-H.02 — exact Turn X/X proof
// -----------------------------------------------------------------------------

// Purpose: Hero checked Turn exactly once and never called/bet/raised/jammed.
// Source/Provenance: T closed round-3 bot counters.
##f$cc_hist_river_probe_hero_checked_turn_clean##
When !IsRiver Return false Force
When didchecround3 != 1 Return false Force
When didcallround3 != 0 Return false Force
When didraisround3 != 0 Return false Force
When didbetsizeround3 != 0 Return false Force
When didalliround3 != 0 Return false Force
When Others Return true Force

// Purpose: nobody aggressed Turn. This is what separates River Probe from the
// Gate09 River-Donk parent, which requires a Turn X/C against an aggressor.
// Source/Provenance: T OpenHoldem raisbits3.
##f$cc_hist_river_probe_no_turn_aggressor##
IsRiver && raisbits3 = 0

// Purpose: canonical closed Turn check-through after actual flop X/C.
// Source/Provenance: T/A composition.
##f$cc_hist_river_probe_turn_checkthrough_clean##
f$cc_hist_river_probe_hero_checked_turn_clean &&
f$cc_hist_river_probe_no_turn_aggressor

// -----------------------------------------------------------------------------
// RP-H.03 — preflop scope without false initiative exclusion
// -----------------------------------------------------------------------------

// Purpose: Gate11 currently recognizes audited pot families through clean 4BP.
// Framework ownership itself is initiative-agnostic once Hero called flop; 5bet+
// chronology remains outside current CashCrusher postflop certification.
// Source/Provenance: T/A Framework BotCalledOnFlop route + P 5bet+ boundary.
##f$cc_river_probe_preflop_context_supported##
f$cc_pot_family_id >= 1 && f$cc_pot_family_id <= 4

// Purpose: audit metadata exposing the source-fidelity correction: PFA/3bettor/
// 4bettor are NOT automatically excluded after an actual flop X/C.
// Source/Provenance: A exact Framework ownership fact; descriptor only.
##f$cc_river_probe_preflop_aggressor_history_allowed##
f$cc_pf_role_pfa || f$cc_pf_role_3bettor || f$cc_pf_role_4bettor

// Purpose: exact LAST does not belong to OOP/early River Probe.
// Source/Provenance: A legacy positional router boundary.
##f$cc_river_probe_excluded_last##
f$cc_relative_postflop_pos_id = 3

// -----------------------------------------------------------------------------
// RP-H.04 — canonical River opportunity
// -----------------------------------------------------------------------------

// Purpose: Hero's first River action with no wager facing.
// Source/Provenance: T current street state.
##f$cc_river_probe_first_river_action_clean##
IsRiver &&
BotsActionsOnThisRoundIncludingChecks = 0 &&
AmountToCall = 0 &&
balance > 0

// Purpose: canonical River Probe opportunity after exact X/C -> X/X history.
// Preflop role is intentionally NOT an ownership filter; strategic descendants
// must still prove their own pot/range parent before betting.
// Source/Provenance: T/A/P exact history/position composition.
##f$cc_river_probe_base_opportunity##
When !f$cc_context_valid Return false Force
When !f$cc_river_probe_first_river_action_clean Return false Force
When !f$cc_hist_river_probe_flop_checkcall_clean Return false Force
When !f$cc_hist_river_probe_turn_checkthrough_clean Return false Force
When !f$cc_river_probe_preflop_context_supported Return false Force
When f$cc_river_probe_excluded_last Return false Force
When f$cc_relative_postflop_pos_id != 1 && f$cc_relative_postflop_pos_id != 2 Return false Force
When Others Return true Force

// Purpose: current-HU River Probe opportunity. The sole live opponent must be the
// actual flop aggressor; otherwise an old folded actor could leak into the node.
// Source/Provenance: T/A exact actor ownership.
##f$cc_river_probe_hu_opportunity##
f$cc_river_probe_base_opportunity &&
f$cc_hu &&
f$cc_hu_oop &&
headsupchair = lastraised2

// Purpose: current multiway opportunities are labeled but remain strategy-owned
// only after an explicit Gate11 descendant audits actor/order/range provenance.
// Source/Provenance: P gap metadata.
##f$cc_river_probe_multiway_opportunity##
f$cc_river_probe_base_opportunity && f$cc_multiway

// Purpose: compact current field origin: 1 true HU, 2 preflop-reduced HU,
// 3 postflop-reduced HU after flop X/C, 4 current multiway.
// Source/Provenance: P audit metadata; does not choose strategy.
##f$cc_river_probe_field_origin_id##
When !f$cc_river_probe_base_opportunity Return 0 Force
When f$cc_river_probe_hu_opportunity && f$cc_true_hu Return 1 Force
When f$cc_river_probe_hu_opportunity && f$cc_hu_origin_preflop_reduced Return 2 Force
When f$cc_river_probe_hu_opportunity && f$cc_hu_origin_postflop_reduced Return 3 Force
When f$cc_river_probe_multiway_opportunity Return 4 Force
When Others Return 0 Force

// Purpose: ownership sanity and anti-node leakage.
// Source/Provenance: P fail-closed invariant.
##f$cc_river_probe_history_consistent##
When f$cc_river_probe_base_opportunity && f$cc_river_probe_excluded_last Return false Force
When f$cc_river_probe_hu_opportunity && headsupchair != lastraised2 Return false Force
When f$cc_river_probe_base_opportunity && !f$cc_river_probe_preflop_context_supported Return false Force
When f$cc_river_probe_base_opportunity && f$cc_river_probe_field_origin_id <= 0 Return false Force
When Others Return true Force
