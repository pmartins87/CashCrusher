// Gate 00E/00F — true multiway context descriptors for 3+ players post-flop
// ============================================================================
// Purpose:
//   Preserve the exact mechanical origin/composition of multiway pots so later
//   policy can decide whether a DeepCrusher 3-way ancestor is A-useful or a new
//   P professional-theory parent is required.
//
// Exact context keeps:
//   deal size + flop-entry count + CURRENT player count + pot family + Hero
//   preflop role + Hero canonical position + live-opponent mask + relative pos.
//
// Important:
//   A turn that is currently 3-way after a 4-way flop is not the same range state
//   as a hand that reached flop 3-way.  Flop-entry provenance survives folds.
//
// Source/Provenance:
//   A = generalization of audited DeepCrusher true-3way positional architecture.
//   P = four-way+ and exact six-max composition architecture where no literal
//       Crusher source exists.  This file itself contains NO betting policy.
// ============================================================================

// Purpose: current street still has at least three live players.
// Source/Provenance: OpenPPL player count, T/A mechanical descriptor.
##f$cc_multiway##
nplayersplaying >= 3 && nplayersplaying <= 6

// Purpose: Hero is first to act among current live players in a multiway pot.
// Source/Provenance: OpenPPL relative position + A multiway alias.
##f$cc_multiway_rel_first##
f$cc_multiway && f$cc_relpos_id = 1

// Purpose: Hero has at least one live opponent before and after in action order.
// Source/Provenance: A relative-position alias; exact players remain in mask.
##f$cc_multiway_rel_middle##
f$cc_multiway && f$cc_relpos_id = 2

// Purpose: Hero is last to act among current live players.
// Source/Provenance: A relative-position alias.
##f$cc_multiway_rel_last##
f$cc_multiway && f$cc_relpos_id = 3

// Purpose: compact role+relative-position ID for diagnostics/coarse routing.
// Source/Provenance: P engineering metadata over A mechanical role/position.
// Safety: not a range/strategy ID and returns 0 outside valid multiway role/pos.
##f$cc_multiway_role_pos_id##
When !f$cc_multiway Return 0 Force
When f$cc_pf_role_id <= 0 Return 0 Force
When f$cc_relpos_id <= 0 Return 0 Force
When Others Return (f$cc_pf_role_id * 10 + f$cc_relpos_id) Force

// Purpose: stable exact multiway mechanical key for logging/fixtures/audits.
// Decimal layout: D F P C R H MM Q where D=dealt, F=flop entrants, P=current,
// C=pot family, R=Hero role, H=Hero pos, MM=opponent mask, Q=relative position.
// Source/Provenance: P engineering ID built entirely from A/T mechanical inputs.
// Safety: this is deliberately NOT a solved strategy lookup key; invalid fields ->0.
##f$cc_multiway_exact_context_id##
When !f$cc_multiway Return 0 Force
When !f$cc_live_mask_consistent Return 0 Force
When f$cc_deal_size <= 0 Return 0 Force
When f$cc_flop_entry_count <= 0 Return 0 Force
When f$cc_pot_family_id <= 0 Return 0 Force
When f$cc_pf_role_id <= 0 Return 0 Force
When f$cc_hero_pos_id <= 0 Return 0 Force
When f$cc_relpos_id <= 0 Return 0 Force
When Others Return ((f$cc_deal_size * 100000000) + (f$cc_flop_entry_count * 10000000) + (nplayersplaying * 1000000) + (f$cc_pot_family_id * 100000) + (f$cc_pf_role_id * 10000) + (f$cc_hero_pos_id * 1000) + (f$cc_opp_live_mask * 10) + f$cc_relpos_id) Force

// Purpose: current multiway PFA acting first.
// Source: DeepCrusher distinguishes first/middle/last in true 3-way attack trees.
// Provenance: A structural parent only; exact ranges/policy remain downstream.
##f$cc_mw_pfa_first##
f$cc_multiway && f$cc_pf_role_pfa && f$cc_multiway_rel_first

// Purpose: current multiway PFA acting from middle relative position.
// Source/Provenance: A structural parent.
##f$cc_mw_pfa_middle##
f$cc_multiway && f$cc_pf_role_pfa && f$cc_multiway_rel_middle

// Purpose: current multiway PFA acting last.
// Source/Provenance: A structural parent; strongest analogy to legacy BTN-v-2p.
##f$cc_mw_pfa_last##
f$cc_multiway && f$cc_pf_role_pfa && f$cc_multiway_rel_last

// Purpose: ordinary-SRP caller acting first in current multiway pot.
// Source/Provenance: A role/position decomposition; no action implied.
##f$cc_mw_srp_caller_first##
f$cc_multiway && f$cc_pf_role_srp_caller && f$cc_multiway_rel_first

// Purpose: ordinary-SRP caller acting middle.
// Source/Provenance: A mechanical/strategic parent only.
##f$cc_mw_srp_caller_middle##
f$cc_multiway && f$cc_pf_role_srp_caller && f$cc_multiway_rel_middle

// Purpose: ordinary-SRP caller acting last.
// Source/Provenance: A mechanical/strategic parent only.
##f$cc_mw_srp_caller_last##
f$cc_multiway && f$cc_pf_role_srp_caller && f$cc_multiway_rel_last

// Purpose: 3bettor acting first in a current multiway pot.
// Source: no complete DeepCrusher deep-cash 3BP strategy; role itself is mechanical.
// Provenance: A/P context parent, strategy later P-heavy.
##f$cc_mw_3bettor_first##
f$cc_multiway && f$cc_pf_role_3bettor && f$cc_multiway_rel_first

// Purpose: 3bettor acting from middle.
// Source/Provenance: A/P context parent only.
##f$cc_mw_3bettor_middle##
f$cc_multiway && f$cc_pf_role_3bettor && f$cc_multiway_rel_middle

// Purpose: 3bettor acting last.
// Source/Provenance: A/P context parent only.
##f$cc_mw_3bettor_last##
f$cc_multiway && f$cc_pf_role_3bettor && f$cc_multiway_rel_last

// Purpose: any 3bet caller (opener-call or cold-call role) acting first.
// Source/Provenance: P/A role family; exact caller type remains available and
// should be separated again when a policy needs its different range.
##f$cc_mw_3betcaller_first##
f$cc_multiway && (f$cc_pf_role_open_call_3bet || f$cc_pf_role_cold_call_3bet) && f$cc_multiway_rel_first

// Purpose: any 3bet caller acting middle.
// Source/Provenance: P/A coarse parent; not a permission to merge exact ranges.
##f$cc_mw_3betcaller_middle##
f$cc_multiway && (f$cc_pf_role_open_call_3bet || f$cc_pf_role_cold_call_3bet) && f$cc_multiway_rel_middle

// Purpose: any 3bet caller acting last.
// Source/Provenance: P/A coarse parent; exact role still queryable separately.
##f$cc_mw_3betcaller_last##
f$cc_multiway && (f$cc_pf_role_open_call_3bet || f$cc_pf_role_cold_call_3bet) && f$cc_multiway_rel_last

// Purpose: identify a true three-player-flop shape where Hero is last, matching
// the positional shell of legacy 3wBTNv2p.
// Source/Provenance: A legacy ancestry hint only; frequencies are not copied.
// Safety: requires exactly 3 flop entrants, so a later-reduced 4-way flop is excluded.
##f$cc_legacy_mw_ancestor_btnv2p_shape##
f$cc_threeway && f$cc_flop_entry_count = 3 && f$cc_multiway_rel_last

// Purpose: identify true three-player-flop Hero-first shell of 3wSBv2p.
// Source/Provenance: A ancestry hint only.
##f$cc_legacy_mw_ancestor_sbv2p_shape##
f$cc_threeway && f$cc_flop_entry_count = 3 && f$cc_multiway_rel_first

// Purpose: identify true three-player-flop Hero-middle shell of 3wBBv2p.
// Source/Provenance: A ancestry hint only.
##f$cc_legacy_mw_ancestor_bbv2p_shape##
f$cc_threeway && f$cc_flop_entry_count = 3 && f$cc_multiway_rel_middle

// Purpose: flag any pot whose FLOP origin had four or more players.
// Source: no literal Crusher postflop ancestor exists beyond its true 3-way source.
// Provenance: P requirement for new professional-theory parent even if later folds
// reduce the current player count.
##f$cc_mw_requires_new_theory_parent##
f$cc_flop_entry_count >= 4

// Purpose: final mechanical validity contract for current multiway context.
// Source/Provenance: P engineering guard over Gate00 mechanical checks.
// Safety: exact live-mask consistency and exact context ID are mandatory.
##f$cc_multiway_context_valid##
f$cc_context_valid && f$cc_multiway && f$cc_live_mask_consistent && f$cc_multiway_exact_context_id > 0
