// Source boundary: no dedicated deep-stack 3BP second-barrel tree exists in the

// Purpose: dangerous/completed 3BP turn for one-pair value.
// Source/Provenance: A runout caution + P 3BP application.
##f$cc_turn_plain3bp_dangerous_runout##
f$cc_turn_new_completion || f$cc_turn_four_card_straight_or_flush || f$cc_turn_super_completed

// Purpose: clean plain3BP turn.
// Source/Provenance: A/P complement of dangerous runout.
##f$cc_turn_plain3bp_clean_turn##
!f$cc_turn_plain3bp_dangerous_runout

// Purpose: clean A/K range-pressure turn.
// Source/Provenance: P 3bettor range-advantage selector; no action alone.
##f$cc_turn_plain3bp_high_pressure_turn##
f$cc_turn_plain3bp_clean_turn && f$cc_turn_overcard_to_flop && !f$cc_turn_pairs_flop_rank && (f$cc_turn_is_ace || f$cc_turn_is_king)

// -----------------------------------------------------------------------------
// P3.01 — HU versus ORIGINAL OPENER-CALL
// -----------------------------------------------------------------------------

// Purpose: P-heavy plain3BP second barrel versus original opener-call.
// Source: A generic DeepCrusher second-barrel architecture only; no dedicated
// 3BP frequency source.
// Provenance: P professional deep-stack 3BP fill.
// Logic: robust value broad; clean OP/strong TP remain value; medium TP and good
// draws are IP/selection-sensitive; premium draws continue; narrow IP air uses
// high-pressure turns after a small quality-air flop plan.
// Safety: low natural 3BP SPR influences exact conditions but does not create TP+ jam.
##f$cc_turn_plain3bp_hu_opener_action##
When !f$cc_turn_plain3bp_hu_context_consistent Return false Force
When !f$cc_turn_plain3bp_hu_vs_opener Return false Force
When f$cc_hand_trips_or_better Return true Force
When HaveTwoPair && f$cc_turn_four_card_straight_or_flush && f$cc_turn_cbet_hu_high_spr Return false Force
When HaveTwoPair Return true Force
When f$cc_turn_cbet_overpair && f$cc_turn_plain3bp_dangerous_runout && f$cc_turn_cbet_hu_high_spr Return false Force
When f$cc_turn_cbet_overpair Return true Force
When f$cc_turn_cbet_strong_top_pair && f$cc_turn_plain3bp_dangerous_runout && f$cc_turn_cbet_hu_high_spr Return false Force
When f$cc_turn_cbet_strong_top_pair Return true Force
When f$cc_turn_cbet_medium_top_pair && f$cc_hu_ip && f$cc_turn_plain3bp_clean_turn && !f$cc_turn_plain3bp_hu_strong_selection Return true Force
When f$cc_turn_cbet_medium_top_pair && f$cc_pair_plus_draw && !f$cc_turn_super_completed && !f$cc_turn_plain3bp_hu_postmw_origin Return true Force
When f$cc_turn_cbet_medium_top_pair Return false Force
When f$cc_turn_cbet_weak_top_pair Return false Force
When f$cc_turn_cbet_second_pair Return false Force
When f$cc_turn_cbet_third_or_worse_pair Return false Force
When f$cc_hand_no_made && f$cc_turn_cbet_premium_draw && !f$cc_turn_super_completed Return true Force
When f$cc_hand_no_made && f$cc_turn_cbet_good_draw && f$cc_hu_ip && f$cc_turn_plain3bp_high_pressure_turn && !f$cc_turn_plain3bp_hu_strong_selection Return true Force
When f$cc_hand_no_made && f$cc_real_frontdoor_draw Return false Force
When f$cc_turn_cbet_air && f$cc_hu_ip && !f$cc_turn_plain3bp_hu_strong_selection && user_cc_flop_cbet_had_quality_air && f$cc_hist_flop_cbet_plan_size_id = 1 && f$cc_turn_plain3bp_high_pressure_turn Return true Force
When Others Return false Force

// Purpose: plain3BP opener-call HU turn sizing.
// Source/Provenance: P 33/50/75 map informed by lower-SPR 3BP geometry.
##f$cc_turn_plain3bp_hu_opener_size_id##
When !f$cc_turn_plain3bp_hu_opener_action Return 0 Force
When f$cc_hand_trips_or_better Return f$cc_turn_size_75_id Force
When HaveTwoPair && f$cc_turn_plain3bp_dangerous_runout Return f$cc_turn_size_50_id Force
When HaveTwoPair Return f$cc_turn_size_75_id Force
When f$cc_turn_cbet_overpair Return f$cc_turn_size_50_id Force
When f$cc_turn_cbet_strong_top_pair Return f$cc_turn_size_50_id Force
When f$cc_turn_cbet_medium_top_pair Return f$cc_turn_size_33_id Force
When f$cc_hand_no_made && f$cc_turn_cbet_premium_draw Return f$cc_turn_size_75_id Force
When f$cc_hand_no_made && f$cc_turn_cbet_good_draw Return f$cc_turn_size_50_id Force
When f$cc_turn_cbet_air Return f$cc_turn_size_50_id Force
When Others Return 0 Force
