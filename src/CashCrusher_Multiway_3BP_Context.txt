// Purpose:
//   Preserve how many live flop opponents come from each preflop range origin:
//     opener | pre-3bet cold caller | post-3bet cold caller.
//   Strategy may then split plain 3BP from squeeze without losing exact masks.
//
// Source/Provenance:
//   A = exact masks reconstructed in CashCrusher_3BP_Context.txt.
//   P = multiway count/composition routing absent from DeepCrusher.
// ============================================================================

// Purpose: exact unchanged-field multiway 3BP/squeeze PFA flop context.
// Context: 3-6 players entered and still remain; Hero final 3bettor; supported
// first-orbit 3bet chronology; plain or squeeze subtype proven; no flop bet.
// Source/Provenance: A exact preflop mechanics + P multiway routing.
// Safety: unsupported/reversed 3bet chronology and 4BP are excluded.
##f$cc_mw_3bp_pfa_context##
f$cc_multiway_context_valid &&
f$cc_flop_entry_count = nplayersplaying &&
f$cc_flop_entry_count >= 3 &&
f$cc_flop_entry_count <= 6 &&
f$cc_pot_family_id = 3 &&
f$cc_pf_role_3bettor &&
f$cc_pf_3bet_order_supported &&
(f$cc_pf_3bet_plain_proven || f$cc_pf_squeeze_proven) &&
f$cc_flop_cbet_opportunity &&
f$cc_spr_valid

// Purpose: number of live original openers among current opponents (normally 0/1).
// Source/Provenance: A BitCount over exact opener-live mask.
##f$cc_mw_3bp_live_opener_count##
When !f$cc_mw_3bp_pfa_context Return 0 Force
When Others Return BitCount(f$cc_mw_3bp_live_opener_mask) Force

// Purpose: number of live pre-3bet cold callers who survived a squeeze.
// Source/Provenance: A BitCount over exact pre-3bet-caller live mask.
##f$cc_mw_3bp_live_pre3bet_coldcaller_count##
When !f$cc_mw_3bp_pfa_context Return 0 Force
When Others Return BitCount(f$cc_mw_3bp_live_pre3bet_coldcaller_mask) Force

// Purpose: number of live post-3bet cold callers.
// Source/Provenance: A BitCount over exact post-3bet-caller live mask.
##f$cc_mw_3bp_live_post3bet_coldcaller_count##
When !f$cc_mw_3bp_pfa_context Return 0 Force
When Others Return BitCount(f$cc_mw_3bp_live_post3bet_coldcaller_mask) Force

// Purpose: total live opponents whose 3BP range origin is explicitly classified.
// Source/Provenance: P safety aggregation over A exact counts.
##f$cc_mw_3bp_classified_opp_count##
When !f$cc_mw_3bp_pfa_context Return 0 Force
When Others Return (f$cc_mw_3bp_live_opener_count + f$cc_mw_3bp_live_pre3bet_coldcaller_count + f$cc_mw_3bp_live_post3bet_coldcaller_count) Force

// Purpose: exact composition consistency guard before multiway 3BP policy.
// Source/Provenance: P fail-closed invariant.
##f$cc_mw_3bp_context_consistent##
When !f$cc_mw_3bp_pfa_context Return false Force
When f$cc_mw_3bp_classified_opp_count != (f$cc_flop_entry_count - 1) Return false Force
When f$cc_pf_3bet_plain_proven && f$cc_mw_3bp_live_pre3bet_coldcaller_count > 0 Return false Force
When Others Return true Force

// Purpose: exact 3-player multiway 3BP/squeeze flop.
// Source/Provenance: T/A count alias over validated context.
##f$cc_mw_3bp_exact3_context##
f$cc_mw_3bp_pfa_context && f$cc_flop_entry_count = 3 && f$cc_mw_3bp_context_consistent

// Purpose: exact 4-player multiway 3BP/squeeze flop.
// Source/Provenance: T/A count alias.
##f$cc_mw_3bp_exact4_context##
f$cc_mw_3bp_pfa_context && f$cc_flop_entry_count = 4 && f$cc_mw_3bp_context_consistent

// Purpose: exact 5-player multiway 3BP/squeeze flop.
// Source/Provenance: T/A count alias.
##f$cc_mw_3bp_exact5_context##
f$cc_mw_3bp_pfa_context && f$cc_flop_entry_count = 5 && f$cc_mw_3bp_context_consistent

// Purpose: exact 6-player multiway 3BP/squeeze flop.
// Source/Provenance: T/A count alias.
##f$cc_mw_3bp_exact6_context##
f$cc_mw_3bp_pfa_context && f$cc_flop_entry_count = 6 && f$cc_mw_3bp_context_consistent

// Purpose: exact-threeway PLAIN 3BP context.
// Source/Provenance: P strategy parent over A subtype proof.
##f$cc_mw_plain3bp_exact3_context##
f$cc_mw_3bp_exact3_context && f$cc_pf_3bet_plain_proven

// Purpose: exact-threeway SQUEEZE context.
// Source/Provenance: P strategy parent over A squeeze proof.
##f$cc_mw_squeeze_exact3_context##
f$cc_mw_3bp_exact3_context && f$cc_pf_squeeze_proven

// Purpose: four-way+ PLAIN 3BP context.
// Source/Provenance: P strategy parent; exact count remains available.
##f$cc_mw_plain3bp_4plus_context##
f$cc_mw_3bp_pfa_context && f$cc_flop_entry_count >= 4 && f$cc_pf_3bet_plain_proven && f$cc_mw_3bp_context_consistent

// Purpose: four-way+ SQUEEZE context.
// Source/Provenance: P strategy parent; exact count remains available.
##f$cc_mw_squeeze_4plus_context##
f$cc_mw_3bp_pfa_context && f$cc_flop_entry_count >= 4 && f$cc_pf_squeeze_proven && f$cc_mw_3bp_context_consistent

// Purpose: multiway 3BP Hero acts FIRST.
// Source/Provenance: A relative position + P strategy parent.
##f$cc_mw_3bp_first_context##
f$cc_mw_3bp_pfa_context && f$cc_multiway_rel_first

// Purpose: multiway 3BP Hero acts MIDDLE.
// Source/Provenance: A relative position + P strategy parent.
##f$cc_mw_3bp_middle_context##
f$cc_mw_3bp_pfa_context && f$cc_multiway_rel_middle

// Purpose: multiway 3BP Hero acts LAST.
// Source/Provenance: A relative position + P strategy parent.
##f$cc_mw_3bp_last_context##
f$cc_mw_3bp_pfa_context && f$cc_multiway_rel_last

// Purpose: original opener is still one of the live opponents.
// Source/Provenance: A exact range-origin flag.
##f$cc_mw_3bp_has_live_opener##
f$cc_mw_3bp_pfa_context && f$cc_mw_3bp_live_opener_count = 1

// Purpose: at least one selected pre-3bet cold caller remains after a squeeze.
// Source/Provenance: P/A composition pressure descriptor.
##f$cc_mw_3bp_has_live_pre3bet_coldcaller##
f$cc_mw_3bp_pfa_context && f$cc_mw_3bp_live_pre3bet_coldcaller_count > 0

// Purpose: at least one post-3bet cold caller remains.
// Source/Provenance: P/A composition pressure descriptor.
##f$cc_mw_3bp_has_live_post3bet_coldcaller##
f$cc_mw_3bp_pfa_context && f$cc_mw_3bp_live_post3bet_coldcaller_count > 0

// Purpose: field contains no opener and is entirely coldcaller-origin ranges.
// Source/Provenance: P selected/condensed broad composition flag.
##f$cc_mw_3bp_all_live_are_coldcallers##
f$cc_mw_3bp_pfa_context && !f$cc_mw_3bp_has_live_opener && (f$cc_mw_3bp_live_pre3bet_coldcaller_count + f$cc_mw_3bp_live_post3bet_coldcaller_count) = (f$cc_flop_entry_count - 1)

// Purpose: compact exact audit key retaining flop count, Hero pos, live mask,
// opener/pre/post caller counts and relative position.
// Source/Provenance: P engineering metadata; not a solved strategy key.
##f$cc_mw_3bp_range_key##
When !f$cc_mw_3bp_pfa_context Return 0 Force
When Others Return ((f$cc_flop_entry_count * 100000000) + (f$cc_hero_pos_id * 1000000) + (f$cc_opp_live_mask * 10000) + (f$cc_mw_3bp_live_opener_count * 1000) + (f$cc_mw_3bp_live_pre3bet_coldcaller_count * 100) + (f$cc_mw_3bp_live_post3bet_coldcaller_count * 10) + f$cc_relpos_id) Force
