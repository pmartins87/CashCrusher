// Purpose:
//   Compare the already-reviewed Turn-Donk size with Hero stack and exact HU /
//   shallowest+deepest multiway effective relationships. Geometry only.
//
// Source/Provenance:
//   T = OpenHoldem/OpenPPL current-pot, balance and first-action bet geometry.
//   A = dual-bound multiway correction frozen in earlier attack gates.
//   P = sidepot-safe diagnostics. Historical near-all-in thresholds are retained
//       as review signals, never generic Turn-Donk shove commands.
//
// Safety:
//   `BetMin` is exactly one big blind in a clean NLHE first-bet state. All other
//   sizes are fraction*round-start-pot. No strategy is changed here.
// ============================================================================

// Purpose: prove simple first-action Turn-Donk geometry is clean.
// Source: Gate08 opportunity requires first Hero Turn action and AmountToCall=0.
// Provenance: T exact currentbet/potplayer/stack guards.
##f$cc_turn_donk_stackgeom_clean##
f$cc_turn_donk_runtime_size_ready &&
f$cc_turn_donk_runtime_mapping_valid &&
AmountToCall = 0 &&
currentbet = 0 &&
potplayer = 0 &&
f$cc_round_start_pot_bb > 0 &&
f$cc_hero_stack_round_start_bb > 0

// Purpose: requested Turn-Donk contribution in big blinds.
// Source/Provenance: T exact first-bet arithmetic; BetMin=1bb.
##f$cc_turn_donk_requested_bet_bb##
When !f$cc_turn_donk_stackgeom_clean Return -1 Force
When f$cc_turn_donk_size_id = 5 Return 1.00 Force
When f$cc_turn_donk_requested_pot_fraction <= 0 Return -1 Force
When Others Return (f$cc_turn_donk_requested_pot_fraction * f$cc_round_start_pot_bb) Force

// Purpose: requested Donk / Hero available Turn-start stack.
// Source/Provenance: T mechanical balance ratio.
##f$cc_turn_donk_requested_hero_balance_ratio##
When !f$cc_turn_donk_stackgeom_clean Return -1 Force
When f$cc_hero_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_donk_requested_bet_bb / f$cc_hero_stack_round_start_bb) Force

// Purpose: requested Donk / exact HU effective stack.
// Source/Provenance: T/A HU effective geometry.
##f$cc_turn_donk_requested_hu_effective_ratio##
When !f$cc_turn_donk_stackgeom_clean Return -1 Force
When !f$cc_hu Return -1 Force
When f$cc_hu_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_donk_requested_bet_bb / f$cc_hu_effective_stack_round_start_bb) Force

// Purpose: requested Donk / shallowest current multiway effective relation.
// Source/Provenance: T/A sidepot descriptor only.
##f$cc_turn_donk_requested_mw_shallowest_ratio##
When !f$cc_turn_donk_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_shallowest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_donk_requested_bet_bb / f$cc_mw_shallowest_effective_stack_round_start_bb) Force

// Purpose: requested Donk / deepest all-live current multiway effective relation.
// Source/Provenance: T/A whole-field effective denominator.
##f$cc_turn_donk_requested_mw_deepest_ratio##
When !f$cc_turn_donk_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_deepest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_turn_donk_requested_bet_bb / f$cc_mw_deepest_effective_stack_round_start_bb) Force

// Purpose: reviewed size already exhausts Hero stack.
// Source/Provenance: T mechanical reach.
##f$cc_turn_donk_requested_reaches_hero_stack##
f$cc_turn_donk_requested_hero_balance_ratio >= 1.00

// Purpose: reviewed size reaches exact HU effective relation.
// Source/Provenance: T/A mechanical reach.
##f$cc_turn_donk_requested_reaches_hu_effective##
f$cc_hu && f$cc_turn_donk_requested_hu_effective_ratio >= 1.00

// Purpose: reviewed size reaches at least one shallow multiway relationship.
// Source/Provenance: T/A sidepot diagnostic only.
##f$cc_turn_donk_requested_reaches_mw_shallowest##
f$cc_multiway && f$cc_turn_donk_requested_mw_shallowest_ratio >= 1.00

// Purpose: reviewed size reaches deepest/all-live multiway relationship.
// Source/Provenance: T/A whole-field mechanical reach.
##f$cc_turn_donk_requested_reaches_mw_deepest##
f$cc_multiway && f$cc_turn_donk_requested_mw_deepest_ratio >= 1.00

// Purpose: detect shortest-only sidepot all-in while another opponent remains deep.
// Source/Provenance: P/T sidepot safety diagnostic.
##f$cc_turn_donk_mw_sidepot_divergence_candidate##
f$cc_turn_donk_requested_reaches_mw_shallowest &&
!f$cc_turn_donk_requested_reaches_mw_deepest

// Purpose: historical short-stack near-all-in signal retained for node-by-node
// review rather than blanket acceptance or blanket deletion.
// Source: DeepCrusher ~50/55/60 bet/commitment mechanisms.
// Provenance: A diagnostic under the project's binding review-not-delete rule.
##f$cc_turn_donk_historical_near_allin_review##
When !f$cc_turn_donk_stackgeom_clean Return false Force
When f$cc_turn_donk_requested_hero_balance_ratio >= 0.50 Return true Force
When f$cc_hu && f$cc_turn_donk_requested_hu_effective_ratio >= 0.50 Return true Force
When f$cc_multiway && f$cc_turn_donk_requested_mw_deepest_ratio >= 0.50 Return true Force
When Others Return false Force
