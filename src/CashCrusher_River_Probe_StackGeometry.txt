// Gate 11L.2 — mechanical River-Probe stack-sensitive execution geometry
// ============================================================================
// Purpose:
//   Measure the already-selected River-Probe request against Hero balance and
//   exact HU / dual-bound multiway effective stacks. Historical source-era
//   near-all-in thresholds remain diagnostics, never automatic strategy.
//
// MIN handling:
//   BetMin is a minimum bet, not a pot fraction. Prior runtime audit establishes
//   the minimum as 1bb, so geometry uses 1bb while fraction metadata uses -1.
//
// Source/Provenance:
//   T = OpenHoldem pot/balance/currentbet + Gate00 effective-stack geometry.
//   A = source-era ~50/60 commitment signals retained only for audit visibility.
//   P = dual-bound sidepot safety and whole-field execution-equivalence rules.
//
// Safety:
//   * clean initial River bet only (AmountToCall/currentbet/potplayer = 0);
//   * shortest-only multiway reach is a sidepot event, not whole-field all-in;
//   * deepest effective relation controls all-live equivalence;
//   * no BetMax is returned by this file.
// ============================================================================

// Purpose: simple street-start geometry is valid for this initial River Probe.
// Source/Provenance: T runtime current-bet/pot facts + P reviewed mapping guard.
##f$cc_river_probe_stackgeom_clean##
f$cc_river_probe_runtime_size_ready &&
f$cc_river_probe_runtime_mapping_valid &&
AmountToCall = 0 &&
currentbet = 0 &&
potplayer = 0 &&
f$cc_round_start_pot_bb > 0 &&
f$cc_hero_stack_round_start_bb > 0

// Purpose: exact requested Hero contribution in BB before any Villain response.
// Source/Provenance: T BetMin=1bb + A fixed-fraction initial-bet arithmetic.
##f$cc_river_probe_requested_bet_bb##
When !f$cc_river_probe_stackgeom_clean Return -1 Force
When f$cc_river_probe_size_id = 1 Return 1.00 Force
When f$cc_river_probe_requested_pot_fraction <= 0 Return -1 Force
When Others Return (f$cc_river_probe_requested_pot_fraction * f$cc_round_start_pot_bb) Force

// Purpose: requested River Probe / Hero street-start stack.
// Source/Provenance: T exact mechanical balance ratio.
##f$cc_river_probe_requested_hero_balance_ratio##
When !f$cc_river_probe_stackgeom_clean Return -1 Force
When f$cc_hero_stack_round_start_bb <= 0 Return -1 Force
When f$cc_river_probe_requested_bet_bb < 0 Return -1 Force
When Others Return (f$cc_river_probe_requested_bet_bb / f$cc_hero_stack_round_start_bb) Force

// Purpose: Hero residual stack after requested bet, before response.
// Source/Provenance: T arithmetic diagnostic.
##f$cc_river_probe_hero_residual_after_bet_bb##
When f$cc_river_probe_requested_bet_bb < 0 Return -1 Force
When Others Return (f$cc_hero_stack_round_start_bb - f$cc_river_probe_requested_bet_bb) Force

// Purpose: historical ~50%-Hero-stack review signal only.
// Source/Provenance: A DeepCrusher commitment ancestry; NOT strategy.
##f$cc_river_probe_hist_hero_balance_50_trigger##
f$cc_river_probe_requested_hero_balance_ratio >= 0.50

// Purpose: historical explicit ~60%-Hero-stack review signal only.
// Source/Provenance: A sizing-promotion ancestry; NOT strategy.
##f$cc_river_probe_hist_hero_stack_60_trigger##
f$cc_river_probe_requested_hero_balance_ratio >= 0.60

// Purpose: requested River Probe / exact current HU effective stack.
// Source/Provenance: T/A Gate00 HU street-start effective geometry.
##f$cc_river_probe_requested_hu_effective_ratio##
When !f$cc_river_probe_stackgeom_clean Return -1 Force
When !f$cc_hu Return -1 Force
When f$cc_hu_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_river_probe_requested_bet_bb / f$cc_hu_effective_stack_round_start_bb) Force

// Purpose: historical ~50%-HU-effective review signal only.
// Source/Provenance: A callback ancestry; no action implied.
##f$cc_river_probe_hist_hu_effective_50_trigger##
f$cc_river_probe_requested_hu_effective_ratio >= 0.50

// Purpose: requested River Probe / shallowest current multiway effective relation.
// Source/Provenance: T/A corrected dual-bound multiway geometry.
##f$cc_river_probe_requested_mw_shallowest_effective_ratio##
When !f$cc_river_probe_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_shallowest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_river_probe_requested_bet_bb / f$cc_mw_shallowest_effective_stack_round_start_bb) Force

// Purpose: requested River Probe / deepest all-live multiway effective relation.
// Source/Provenance: T/A whole-field denominator.
##f$cc_river_probe_requested_mw_deepest_effective_ratio##
When !f$cc_river_probe_stackgeom_clean Return -1 Force
When !f$cc_multiway Return -1 Force
When !f$cc_mw_stack_geometry_valid Return -1 Force
When f$cc_mw_deepest_effective_stack_round_start_bb <= 0 Return -1 Force
When Others Return (f$cc_river_probe_requested_bet_bb / f$cc_mw_deepest_effective_stack_round_start_bb) Force

// Purpose: historical ~50%-effective MW review signal using deepest/all-live.
// Source/Provenance: A migration diagnostic only.
##f$cc_river_probe_hist_mw_effective_50_trigger##
f$cc_river_probe_requested_mw_deepest_effective_ratio >= 0.50

// Purpose: request reaches Hero's entire balance.
// Source/Provenance: T mechanical balance geometry.
##f$cc_river_probe_requested_reaches_hero_stack##
f$cc_river_probe_requested_hero_balance_ratio >= 1.00

// Purpose: request reaches exact HU effective relationship.
// Source/Provenance: T exact HU arithmetic.
##f$cc_river_probe_requested_reaches_hu_effective##
f$cc_hu && f$cc_river_probe_requested_hu_effective_ratio >= 1.00

// Purpose: request reaches at least the shallowest MW effective relationship.
// Source/Provenance: T/A sidepot descriptor only.
##f$cc_river_probe_requested_reaches_mw_shallowest##
f$cc_multiway && f$cc_river_probe_requested_mw_shallowest_effective_ratio >= 1.00

// Purpose: request reaches deepest/all-live MW effective relationship.
// Source/Provenance: T/A whole-field mechanical reach.
##f$cc_river_probe_requested_reaches_mw_deepest##
f$cc_multiway && f$cc_river_probe_requested_mw_deepest_effective_ratio >= 1.00

// Purpose: shortest-only all-in/sidepot reach while a deeper opponent remains.
// Source/Provenance: P/T safety diagnostic; never whole-field commitment.
##f$cc_river_probe_mw_sidepot_divergence_candidate##
f$cc_river_probe_requested_reaches_mw_shallowest &&
!f$cc_river_probe_requested_reaches_mw_deepest

// Purpose: stack-sensitive geometry engaged and deserves audit visibility.
// True means review, never an automatic shove.
// Source/Provenance: A/P metadata over T geometry.
##f$cc_river_probe_stack_sensitive_review_required##
When !f$cc_river_probe_stackgeom_clean Return false Force
When f$cc_river_probe_hist_hero_balance_50_trigger Return true Force
When f$cc_hu && f$cc_river_probe_hist_hu_effective_50_trigger Return true Force
When f$cc_multiway && f$cc_river_probe_hist_mw_effective_50_trigger Return true Force
When f$cc_river_probe_requested_reaches_hero_stack Return true Force
When f$cc_hu && f$cc_river_probe_requested_reaches_hu_effective Return true Force
When f$cc_multiway && f$cc_river_probe_requested_reaches_mw_shallowest Return true Force
When Others Return false Force

// Purpose: compact review flags.
// Bits: 1 Hero>=50; 2 Hero>=60; 4 HU-eff>=50; 8 reaches HU effective;
// 16 reaches MW shallowest; 32 MW-deepest>=50; 64 reaches MW deepest;
// 128 shortest-only sidepot divergence; 256 MIN request; 512 exact River30.
// Source/Provenance: P metadata over A/T diagnostics.
##f$cc_river_probe_stack_review_flags_id##
When !f$cc_river_probe_stackgeom_clean Return 0 Force
When Others Return ((f$cc_river_probe_hist_hero_balance_50_trigger * 1) + (f$cc_river_probe_hist_hero_stack_60_trigger * 2) + ((f$cc_hu && f$cc_river_probe_hist_hu_effective_50_trigger) * 4) + ((f$cc_hu && f$cc_river_probe_requested_reaches_hu_effective) * 8) + ((f$cc_multiway && f$cc_river_probe_requested_reaches_mw_shallowest) * 16) + ((f$cc_multiway && f$cc_river_probe_hist_mw_effective_50_trigger) * 32) + ((f$cc_multiway && f$cc_river_probe_requested_reaches_mw_deepest) * 64) + ((f$cc_multiway && f$cc_river_probe_mw_sidepot_divergence_candidate) * 128) + ((f$cc_river_probe_size_id = 1) * 256) + ((f$cc_river_probe_size_id = 7) * 512)) Force

// ============================================================================
// End Gate11L.2 River-Probe stack geometry. No all-in action here.
